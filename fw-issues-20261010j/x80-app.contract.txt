  state log = ""
  state n = 0
  state a1 = false
  state a2 = false
  state a3 = false
  state a4 = false
  state a5 = false
  state b1 = false
  state b2 = false
  state b3 = false
  state b4 = false
  state b5 = false
  action ev(name: string, over: bool)
    n = n + 1
    log = `${n}${name}${over ? "+" : "-"} ${log}`
    a1 = name == "A1" ? over : a1
    a2 = name == "A2" ? over : a2
    a3 = name == "A3" ? over : a3
    a4 = name == "A4" ? over : a4
    a5 = name == "A5" ? over : a5

    b3 = name == "B3" ? over : b3
    b4 = name == "B4" ? over : b4
    b5 = name == "B5" ? over : b5
  action pd(name: string)
    n = n + 1
    log = `${n}${name}v ${log}`
  action pu(name: string)
    n = n + 1
    log = `${n}${name}^ ${log}`
