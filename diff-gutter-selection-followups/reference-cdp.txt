--- ref-textsel.mjs
// diff-gutter-selection-followups GS-1: in the reference's open diff, (1) a gutter drag over line numbers, (2) a drag inside
// the code, (3) a press on a number while code text is selected; after each, what the document's and the diff's shadow
// root's text selection hold. Usage: node ref-textsel.mjs <cdpPort> <fromN> <toN> <shotPrefix>
import { open, log, gutter } from './ref-lib.mjs';
const [port, fromN, toN, shot] = process.argv.slice(2);
const page = await open(port);
const cancel = page.getByRole('button', { name: 'Cancel', exact: true });
const sel = () => page.evaluate(() => {
  const doc = window.getSelection();
  const out = { document: { type: doc.type, rangeCount: doc.rangeCount, text: doc.toString() }, shadow: [] };
  for (const c of document.querySelectorAll('diffs-container')) {
    const root = c.shadowRoot; if (!root || typeof root.getSelection !== 'function') continue;
    const s = root.getSelection(); if (s && s.rangeCount) out.shadow.push({ type: s.type, text: s.toString() });
  }
  // The column-number cells' user-select, and the code's.
  const c = document.querySelector('diffs-container')?.shadowRoot;
  const num = c?.querySelector('[data-column-number]'), line = c?.querySelector('[data-line]');
  out.userSelect = { number: num ? getComputedStyle(num).userSelect : null, numberContent: num?.querySelector('[data-line-number-content]') ? getComputedStyle(num.querySelector('[data-line-number-content]')).userSelect : null, line: line ? getComputedStyle(line).userSelect : null };
  return out;
});
const clear = () => page.evaluate(() => { window.getSelection().removeAllRanges(); for (const c of document.querySelectorAll('diffs-container')) c.shadowRoot?.getSelection?.()?.removeAllRanges(); });
const g = await gutter(page);
const from = g.find(e => e.n === fromN), to = g.find(e => e.n === toN);
log({ from, to });
await clear();
// (1) The gutter drag: press on the number, move down to the other number, release.
await page.mouse.move(from.x + from.w - 8, from.y + 10); await page.waitForTimeout(200);
await page.mouse.down();
for (let i = 1; i <= 8; i++) { await page.mouse.move(from.x + from.w - 8, from.y + 10 + (to.y - from.y) * i / 8); await page.waitForTimeout(40); }
await page.waitForTimeout(300);
log({ step: 'gutter drag held', ...(await sel()) });
await page.screenshot({ path: `${shot}-1-gutter-held.png` });
await page.mouse.up(); await page.waitForTimeout(600);
log({ step: 'gutter drag released', ...(await sel()) });
if (await cancel.count()) { await cancel.first().click(); await page.waitForTimeout(600); }
// (2) A drag inside the code: from 60pt into the code of the first line to 160pt into the code of the last.
const codeX = from.x + from.w + 60;
await page.mouse.move(codeX, from.y + 10); await page.waitForTimeout(200);
await page.mouse.down();
for (let i = 1; i <= 8; i++) { await page.mouse.move(codeX + 100 * i / 8, from.y + 10 + (to.y - from.y) * i / 8); await page.waitForTimeout(40); }
await page.mouse.up(); await page.waitForTimeout(600);
log({ step: 'code drag', ...(await sel()), gutterSelected: (await gutter(page)).filter(e => e.selected).map(e => e.n) });
await page.screenshot({ path: `${shot}-2-code-drag.png` });
// (3) A press on a number while that code selection stands.
await page.mouse.click(from.x + from.w - 8, to.y + 10); await page.waitForTimeout(600);
log({ step: 'number press after code selection', ...(await sel()), gutterSelected: (await gutter(page)).filter(e => e.selected).map(e => e.n) });
await page.screenshot({ path: `${shot}-3-number-press.png` });
if (await cancel.count()) { await cancel.first().click(); await page.waitForTimeout(600); }
process.exit(0);

// GS-1: a drag inside the code from line <fromN> to <toN> (the open diff), then what the selection holds.

--- ref-files-sel2.mjs
// GS-2: the Files surface's committed selection (after the release), light and dark: every painted element in the
// selected rows' band (any non-transparent background in the shadow root), the number cells, and the draft.
// Usage: node ref-files-sel2.mjs <cdpPort> <fromN> <toN> <shotPrefix>
import { open, log } from './ref-lib.mjs';
const [port, fromN, toN, shot] = process.argv.slice(2);
const page = await open(port);
const cancel = page.getByRole('button', { name: 'Cancel', exact: true });
if (await cancel.count()) { await cancel.first().click(); await page.waitForTimeout(600); }
await page.emulateMedia({ colorScheme: 'light' }); await page.waitForTimeout(800);
const files = () => page.evaluate(() => { const out = []; for (const c of document.querySelectorAll('diffs-container')) { const root = c.shadowRoot; if (!root) continue; for (const el of root.querySelectorAll('[data-column-number]')) { const r = el.getBoundingClientRect(); if (r.width > 0) out.push({ n: el.getAttribute('data-column-number'), x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height), sel: el.getAttribute('data-selected-line') }); } } return out; });
const painted = () => page.evaluate(() => {
  const cv = document.createElement('canvas'); cv.width = cv.height = 1; const cx = cv.getContext('2d', { willReadFrequently: true });
  const hex = (c) => { if (!c || c === 'rgba(0, 0, 0, 0)') return c; cx.clearRect(0, 0, 1, 1); cx.fillStyle = '#000'; cx.fillStyle = c; cx.fillRect(0, 0, 1, 1); const d = cx.getImageData(0, 0, 1, 1).data; return '#' + [d[0], d[1], d[2]].map(v => v.toString(16).padStart(2, '0')).join('') + (d[3] < 255 ? d[3].toString(16).padStart(2, '0') : ''); };
  const out = [];
  for (const c of document.querySelectorAll('diffs-container')) {
    const root = c.shadowRoot; if (!root) continue;
    for (const el of root.querySelectorAll('*')) {
      const r = el.getBoundingClientRect(); if (r.width === 0 || r.height === 0 || r.y > 200) continue;
      for (const pseudo of [null, '::before', '::after']) {
        const s = getComputedStyle(el, pseudo);
        const bg = s.backgroundColor, img = s.backgroundImage, shadow = s.boxShadow;
        if ((bg && bg !== 'rgba(0, 0, 0, 0)') || (img && img !== 'none') || (shadow && shadow !== 'none')) {
          if (pseudo && s.content === 'none') continue;
          out.push({ tag: el.tagName.toLowerCase(), pseudo, attrs: [...el.attributes].map(a => `${a.name}=${a.value}`).join(' ').slice(0, 140), x: +r.x.toFixed(1), y: +r.y.toFixed(1), w: +r.width.toFixed(1), h: +r.height.toFixed(1), bg: hex(bg), img: img === 'none' ? undefined : img.slice(0, 120), shadow: shadow === 'none' ? undefined : shadow, color: hex(s.color), pw: pseudo ? s.width : undefined, pleft: pseudo ? s.left : undefined, pos: pseudo ? s.position : undefined });
        }
      }
    }
  }
  return out;
});
const g = await files();
const from = g.find(e => e.n === fromN), to = g.find(e => e.n === toN);
await page.mouse.move(from.x + from.w - 8, from.y + 10); await page.waitForTimeout(300);
await page.mouse.down();
for (let i = 1; i <= 6; i++) { await page.mouse.move(from.x + from.w - 8, from.y + 10 + (to.y - from.y) * i / 6); await page.waitForTimeout(40); }
await page.waitForTimeout(200);
log({ phase: 'held', scheme: 'light', gutter: await files(), painted: await painted() });
await page.mouse.up(); await page.waitForTimeout(1500);
for (const scheme of ['light', 'dark']) {
  await page.emulateMedia({ colorScheme: scheme }); await page.waitForTimeout(1500);
  log({ phase: 'released', scheme, gutter: await files(), painted: await painted(), text: await page.evaluate(() => getSelection().toString()) });
  await page.screenshot({ path: `${shot}-${scheme}.png` });
}
await page.emulateMedia({ colorScheme: 'light' }); await page.waitForTimeout(500);
process.exit(0);
