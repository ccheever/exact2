=== ref-split.mjs
// diff-gutter-selection-followups review fix: the reference's Diff panel in split view (the open diff), nothing selected,
// then a drag over the additions side's numbers <fromN> to <toN> (held, then released and the draft cancelled), and what
// the text selection holds. Usage: node ref-split.mjs <cdpPort> <fromN> <toN> <shotPrefix>
import { open, log, gutter } from '/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-audit/lanes/diff-gutter-selection-followups/ref-lib.mjs';
const [port, fromN, toN, shot] = process.argv.slice(2);
const page = await open(port);
const cancel = page.getByRole('button', { name: 'Cancel', exact: true });
const sel = () => page.evaluate(() => {
  const doc = window.getSelection(), shadow = [];
  for (const c of document.querySelectorAll('diffs-container')) { const s = c.shadowRoot?.getSelection?.(); if (s && s.rangeCount) shadow.push({ type: s.type, text: s.toString() }); }
  return { document: { type: doc.type, text: doc.toString() }, shadow };
});
await page.getByRole('radio', { name: 'Split diff view' }).or(page.getByRole('button', { name: 'Split diff view' })).first().click();
await page.waitForTimeout(2500);
await page.evaluate(() => { window.getSelection().removeAllRanges(); });
const g = await gutter(page);
log({ gutter: g.map(e => [e.n, e.type, e.x, e.y, e.w]) });
await page.screenshot({ path: `${shot}-1-open.png` });
// The additions side: the rightmost cell of each number.
const right = (n) => g.filter(e => e.n === n).sort((a, b) => b.x - a.x)[0];
const from = right(fromN), to = right(toN);
log({ from, to });
await page.mouse.move(from.x + from.w - 8, from.y + 10); await page.waitForTimeout(200);
await page.mouse.down();
for (let i = 1; i <= 8; i++) { await page.mouse.move(from.x + from.w - 8, from.y + 10 + (to.y - from.y) * i / 8); await page.waitForTimeout(40); }
await page.waitForTimeout(300);
log({ step: 'split gutter drag held', ...(await sel()), selected: (await gutter(page)).filter(e => e.selected !== null).map(e => [e.n, e.x]) });
await page.screenshot({ path: `${shot}-2-gutter-held.png` });
await page.mouse.up(); await page.waitForTimeout(600);
log({ step: 'released', ...(await sel()) });
if (await cancel.count()) { await cancel.first().click(); await page.waitForTimeout(600); }
process.exit(0);
