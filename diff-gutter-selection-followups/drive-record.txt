#!/bin/zsh
# One agent drive of the clone for diff-gutter-selection-followups. Usage: drive.sh before|after
# before: the feature tip 327336447 built in t3-code-diff-gutter-selection-followups-before (the shared evidence base
# c03d7e908 predates #413's drag, whose gutter selects text); after: this worktree's bundle. Same steps, 1280x840.
# (1) The fixture thread's Diff panel (Changes), .agents/skills/verify/SKILL.md: numbers 2 to 4 dragged (held, then
# released), a drag in the code from line 2 to line 4, then a press on number 3. (2) The Files surface: fixture.txt,
# number 1 pressed and number 2 Shift-pressed, light then dark. (3) The GitHub lane's server paired (a fresh single-use
# link; the embedded server stays focused), #168's Code tab, docs/usage.md: numbers 2 to 4 dragged and held (the release
# opens a draft, then Cancel), then a drag in the code from line 2 to line 4.
MODE=$1
A=/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-audit
W=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-diff-gutter-selection-followups
S=$W/target/dgs-drive/shots/$MODE; mkdir -p $S
if [ "$MODE" = before ]; then ROOTDIR=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-diff-gutter-selection-followups-before; LANE=diff-gutter-selection-followups-before; export CLONE_STORAGE=audit-dgs-before-final; else ROOTDIR=$W; LANE=diff-gutter-selection-followups; export CLONE_STORAGE=audit-dgs-after-final; fi
# Both lanes point Codex and Claude at missing binaries (clone-t3-home/userdata/settings.json), as the reference finds neither.
G=$W/examples/t3-code/tools/github-lane
( cd $G && PATH=$HOME/.bun-1.4.2/bin:$PATH T3_GITHUB_LANE_SHARED=/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-ui-parity/github-lane bun lane.mjs pair primary >/dev/null )
PAIR=$(cat $W/target/github-lane/servers/primary/pairing-url); rm -f $W/target/github-lane/servers/primary/pairing-url
D=.agents/skills/verify/SKILL.md
F=docs/usage.md
ops=(
  "clock +20000 real" "clock +3000 real"
  # (1) The thread's Diff panel.
  "tap thread-verify-timeline" "clock +2500 real"
  "tap details-changes" "clock +10000 real"
  # The fresh storage's Nightly mobile-app notice sits over the panel: closed first.
  "tap \"Dismiss notification\"" "clock +1500 real"
  "tap \"Expand $D\"" "clock +4000 real"
  "screenshot $S/1-diff-open.png window"
  "tap diff-line-$D-additions-2 drag to diff-line-$D-additions-4 mouse over 400 during \"clock +1500 real\" \"screenshot $S/2-diff-gutter-held.png window\""
  "clock +1500 real"
  "screenshot $S/3-diff-gutter-released.png window"
  "tap diff-line-$D-additions-2 drag 140 40 from 70 10 mouse over 400" "clock +1500 real"
  "screenshot $S/4-diff-code-drag.png window"
  "tap diff-line-$D-additions-3" "clock +1500 real"
  "screenshot $S/5-diff-number-press.png window"
  # (2) The Files surface.
  "tap panel-add-surface" "clock +1500 real"
  "tap add-surface-files" "clock +4000 real"
  "tap tree-fixture.txt" "clock +4000 real"
  "tap file-line-1" "clock +1000 real"
  "tap file-line-2 modifiers Shift" "clock +1500 real"
  "screenshot $S/6-files-light.png window"
  "tree file-line-1"
  "prefer prefers-color-scheme dark" "clock +1500 real"
  "screenshot $S/7-files-dark.png window"
  "prefer prefers-color-scheme light" "clock +1000 real"
  # (3) Pair the GitHub lane's server, then #168's Code tab.
  "tap connection-settings" "clock +1500 real"
  "tap settings-connections" "clock +1500 real"
  "tap manage-connection" "clock +1000 real"
  "type server-origin $PAIR" "clock +500 real"
  "tap connect" "clock +12000 real"
  "type t3-code key Escape" "clock +1500 real"
  "tap open-pull-requests" "clock +6000 real"
  "type pull-requests-search #168" "clock +6000 real"
  "tap pr-row-168" "clock +6000 real"
  "tap pull-request-tab-code" "clock +15000 real"
  "tap pull-request-code-tree-toggle" "clock +1500 real"
  "tap pull-request-code-tree-more" "clock +6000 real"
  "tap pull-request-code-tree-more" "clock +6000 real"
  "tap pull-request-code-tree-more" "clock +6000 real"
  "tap pull-request-code-tree-toggle" "clock +1500 real"
  "tap pull-request-code-list into file:$F block start" "clock +1500 real"
  "tap pull-request-code-fold-$F" "clock +4000 real"
  "tap pull-request-code-list into file:$F block start" "clock +1500 real"
  "screenshot $S/8-pr-open.png window"
  "tap diff-line-$F-additions-2 drag to diff-line-$F-additions-4 mouse over 400 during \"clock +1500 real\" \"screenshot $S/9-pr-gutter-held.png window\""
  "clock +2000 real"
  "tap pull-request-comment-cancel" "clock +1500 real"
  "tap pull-request-code-list into file:$F block start" "clock +1500 real"
  "tap diff-line-$F-additions-2 drag 140 40 from 70 10 mouse over 400" "clock +1500 real"
  "screenshot $S/10-pr-code-drag.png window"
  "state"
  "logs"
)
print -rl -- "${ops[@]//$PAIR/<pairing link>}" > $S/ops.txt
EXACT_ROOT=$ROOTDIR $A/clone-drive.sh $LANE 16780 --epoch 2026-10-11T09:00:00Z --time-zone Asia/Seoul "${ops[@]}" > $S/drive.jsonl 2> $S/drive.err
echo "exit $?"
