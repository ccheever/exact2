#!/bin/zsh
# browser-surface-capture, review round of 2026-10-10 (second): a drag that starts on a pill button (agent mode, window
# images; not committed). Usage: drive.sh <run name> <before|after>
#   before: the bundle of this PR's last drive (after2, the code of 85ea13fbe), kept in target/fix1010d/before-dev (EXACT_MAC_BIN)
#   after:  this worktree's build
# Ports (lane base 16700): the embedded server 16702, the fixture 16703 (started by the caller).
# Provider state equals the reference lane's: no provider (both CLIs pinned to a missing path).
W=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-browser-surface-capture
F=$W/target/fix1010d
L=$W/target/capture-lane/lane
AL=/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-audit/lanes/browser-surface-capture
P=${1:?run}; MODE=${2:?before|after}; OUT=$F/runs/$P; rm -rf $OUT; mkdir -p $OUT
BUN=$HOME/.bun-1.4.2/bin/bun
export EXACT_APP_DIR=$W/examples/t3-code
if [ "$MODE" = before ]; then export EXACT_MAC_BIN="$F/before-dev/standalone/T3 Code (Exact)" EXACT_ASSETS=$F/before-dev/capture; fi
mkdir -p $F/homes
export T3_LOCAL_HOME=$F/homes/$P T3_LOCAL_PORT=16702 T3CODE_TELEMETRY_ENABLED=false
export CODEX_HOME=$AL/codex CLAUDE_CONFIG_DIR=$AL/claude
export XDG_CONFIG_HOME=$AL/xdg/config XDG_DATA_HOME=$AL/xdg/data XDG_STATE_HOME=$AL/xdg/state XDG_CACHE_HOME=$AL/xdg/cache
export T3_REMOTE_OPEN_LOG=$OUT/opened-urls.txt
export PATH=$L/bin:$HOME/.bun-1.4.2/bin:/usr/bin:/bin:/usr/sbin:/sbin
export T3_LOCAL_RUNTIME_DIR=$L/runtime/t3-0.0.46-nightly.20261005.2667-darwin-arm64
rm -rf $T3_LOCAL_HOME; cp -R $AL/clone-t3-home $T3_LOCAL_HOME
cat > $T3_LOCAL_HOME/userdata/settings.json <<JSON
{
  "enableProviderUpdateChecks": false,
  "providers": { "cursor": { "enabled": false }, "grok": { "enabled": false }, "opencode": { "enabled": false } },
  "providerInstances": {
    "claudeAgent": { "driver": "claudeAgent", "enabled": true, "config": { "customModels": [], "binaryPath": "$L/bin/absent/claude", "homePath": "", "launchArgs": "", "autoCompactWindow": "" } },
    "codex": { "driver": "codex", "enabled": true, "config": { "binaryPath": "$L/bin/absent/codex", "homePath": "", "shadowHomePath": "", "launchArgs": "", "customModels": [] } }
  }
}
JSON
waitline() { while [ $(cat $OUT/agent.jsonl 2>/dev/null | wc -l) -lt $1 ]; do sleep 0.2; done; }
S=$OUT/shots; mkdir -p $S
STEP=("clock +16")
OPS=("clock +15000 real" "clock +6000 real"
  "tap toggle-right-panel" "clock +1500 real"
  "tap surface-browser" "clock +3000 real"
  "type browser-url 127.0.0.1:16703" "type browser-url key Enter" "clock +4000 real"
  # the server's "Nightly needs the beta mobile app" notice covers the panel's top right: dismissed
  "tap toast-dismiss-1" "clock +800 real"
  "tap browser-url hover" "clock +600 real"
  "tap browser-float" "clock +2500 real" layout "screenshot $S/01-float.png window"
  # A: a drag (-200, -100) that starts on Open in right panel, in four moves, the clock moving between them
  "tap browser-mini-handle hover" "clock +800 real" "screenshot $S/02-pill.png window"
  "tap browser-mini-restore down" "tap move by -50 -25 over 40" "${STEP[@]}" "tap move by -50 -25 over 40" "${STEP[@]}" "tap move by -50 -25 over 40" "${STEP[@]}" "tap move by -50 -25 over 40" "${STEP[@]}" "tap up"
  "clock +1500 real" layout tree "screenshot $S/03-button-drag.png window"
  # B: a drag (-100, -40) from the pill's bottom padding (the reference's pdrag point: 28 pt in, 2 pt above the bottom)
  "tap browser-mini-page hover" "clock +600 real" "tap browser-mini-handle hover" "clock +800 real"
  "tap browser-mini-handle down at 28 30" "tap move by -25 -10 over 40" "${STEP[@]}" "tap move by -25 -10 over 40" "${STEP[@]}" "tap move by -25 -10 over 40" "${STEP[@]}" "tap move by -25 -10 over 40" "${STEP[@]}" "tap up"
  "clock +1500 real" layout "screenshot $S/04-padding-drag.png window"
  # C: meant as the dot; the pill stays shown (its focused button), so the handle's middle is Pop into separate window: (+50, +50)
  "tap browser-mini-page hover" "clock +800 real"
  "tap browser-mini-handle down" "tap move by 25 25 over 40" "${STEP[@]}" "tap move by 25 25 over 40" "${STEP[@]}" "tap up"
  "clock +1500 real" layout "screenshot $S/05-dot-drag.png window"
  # C2 (added for after4): the focused pill button keeps the pill shown (focus-within), so C's press met Pop into separate
  # window; a click in the page takes the focus, the pill hides, and the dot itself is dragged (+50, +50) in two moves
  "tap browser-mini-page clicks 1 at 20 150" "clock +800 real" "tap browser-mini-page hover" "clock +800 real" layout
  "tap browser-mini-handle down" "tap move by 25 25 over 40" "${STEP[@]}" "tap move by 25 25 over 40" "${STEP[@]}" "tap up"
  "clock +1500 real" layout "screenshot $S/05b-dot-drag.png window"
  # D: the pill from the dot onto Pop into separate window: the pill stays and the button's tooltip shows (row 2)
  "tap browser-mini-page hover" "clock +600 real" "tap browser-mini-handle hover" "clock +800 real"
  "tap browser-mini-window hover" "clock +800 real" tree "screenshot $S/06-pill-tooltip.png window"
  # E: a press on Open in right panel still acts: the page goes back to the panel
  "tap browser-mini-restore" "clock +2500 real" tree logs "screenshot $S/07-restored.png window")
cd $W
$BUN scripts/agent.mjs macos --json --size 1280x840 "${OPS[@]}" > $OUT/agent.jsonl 2> $OUT/agent.err &
AG=$!; echo "agent pid $AG" > $OUT/pids.txt
waitline 1
env -i HOME=$AL/home USER=$USER PATH=/usr/bin:/bin TMPDIR="$TMPDIR" T3CODE_TELEMETRY_ENABLED=false $T3_LOCAL_RUNTIME_DIR/t3 app $AL/projects/work --base-dir $T3_LOCAL_HOME > $OUT/t3-app.out 2>&1; echo "t3 app exit $?" >> $OUT/pids.txt
wait $AG; echo "agent exit $?" >> $OUT/pids.txt
echo DRIVE-DONE
