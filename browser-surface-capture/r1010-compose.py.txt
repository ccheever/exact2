W, H, BAR, GAP, HEAD = 960, 630, 46, 12, 52

def font(size):
    for path in ('/System/Library/Fonts/SFNS.ttf', '/System/Library/Fonts/Helvetica.ttc'):
        try: return ImageFont.truetype(path, size)
        except Exception: pass
    return ImageFont.load_default()

def cell(path, label, note=None):
    image = Image.new('RGB', (W, H + BAR), '#f4f4f5')
    draw = ImageDraw.Draw(image)
    draw.rectangle([0, 0, W - 1, BAR - 1], fill='#18181b')
    draw.text((12, 11), label, fill='white', font=font(20))
    if path:
        image.paste(Image.open(path).convert('RGB').resize((W, H), Image.LANCZOS), (0, BAR))
    else:
        draw.text((24, BAR + H // 2 - 14), note or '', fill='#52525b', font=font(22))
    return image

def grid(name, title, rows):
    columns = max(len(row) for row in rows)
    out = Image.new('RGB', (columns * W + (columns - 1) * GAP, HEAD + len(rows) * (H + BAR) + (len(rows) - 1) * GAP), 'white')
    ImageDraw.Draw(out).text((12, 12), title, fill='#18181b', font=font(26))
    for r, row in enumerate(rows):
        for c, spec in enumerate(row):
            out.paste(cell(*spec), (c * (W + GAP), HEAD + r * (H + BAR + GAP)))
    out.save(f'{OUT}/{name}', optimize=True)
    print(name, out.size)
