import sys
from PIL import Image, ImageDraw, ImageFont

F = '/Users/daehyeonmun/orca/workspaces/exact2/t3-code-browser-surface-capture/target/fix1010c'
AFTER_RUN = sys.argv[1] if len(sys.argv) > 1 else 'after2'
B = f'{F}/runs/before-a/shots'      # the bundle realinput-1010c drove (a63794adb), the same agent steps
A = f'{F}/runs/{AFTER_RUN}/shots'   # this round's build (drive after2)
A1 = f'{F}/runs/after1/shots'      # drive after1: rows 1 and 7 (after2 stopped on the driver's clock before them; the builds differ only in the pill's tooltip alignment)
R = f'{F}/ref'                      # T3 Code 0.0.45 (1e2ecbd975) over CDP, the same lane data
OUT = f'{F}/evidence'
W, H, BAR, GAP, HEAD = 960, 630, 46, 12, 52

def font(size):
    for path in ('/System/Library/Fonts/SFNS.ttf', '/System/Library/Fonts/Helvetica.ttc'):
        try: return ImageFont.truetype(path, size)
        except Exception: pass
    return ImageFont.load_default()

def cell(path, label):
    image = Image.new('RGB', (W, H + BAR), '#f4f4f5')
    draw = ImageDraw.Draw(image)
    draw.rectangle([0, 0, W - 1, BAR - 1], fill='#18181b')
    draw.text((12, 11), label, fill='white', font=font(20))
    image.paste(Image.open(path).convert('RGB').resize((W, H), Image.LANCZOS), (0, BAR))
    return image

def grid(name, title, rows):
    columns = max(len(row) for row in rows)
    out = Image.new('RGB', (columns * W + (columns - 1) * GAP, HEAD + len(rows) * (H + BAR) + (len(rows) - 1) * GAP), 'white')
    ImageDraw.Draw(out).text((12, 12), title, fill='#18181b', font=font(26))
    for r, row in enumerate(rows):
        for c, (path, label) in enumerate(row):
            out.paste(cell(path, label), (c * (W + GAP), HEAD + r * (H + BAR + GAP)))
    out.save(f'{OUT}/{name}', optimize=True)
    print(name, out.size)
