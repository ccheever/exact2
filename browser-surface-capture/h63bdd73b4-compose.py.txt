W, H, BAR, GAP = 1120, 735, 56, 16

def font(size):
    for path in ('/System/Library/Fonts/SFNS.ttf', '/System/Library/Fonts/Helvetica.ttc'):
        try: return ImageFont.truetype(path, size)
        except Exception: pass
    return ImageFont.load_default()

def compose(name, panels):
    out = Image.new('RGB', (len(panels) * W + (len(panels) - 1) * GAP, H + BAR), 'white')
    draw = ImageDraw.Draw(out)
    for index, (path, label) in enumerate(panels):
        image = Image.open(path).convert('RGB').resize((W, H), Image.LANCZOS)
        x = index * (W + GAP)
        out.paste(image, (x, BAR))
        draw.rectangle([x, 0, x + W - 1, BAR - 1], fill='#18181b')
        draw.text((x + 16, 14), label, fill='white', font=font(26))
    out.save(f'{OUT}/{name}', optimize=True)
    print(name, out.size)
