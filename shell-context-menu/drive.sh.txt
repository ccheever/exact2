else ROOT=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-shell-context-menu; LANE=shell-context-menu; fi
case_() { OPS+=("tap t3-code contextmenu at $1 $2" "clock +600 real" "logs"); }
OPS=("clock +20000 real" "clock +3000 real" "tap thread-verify-timeline" "clock +3000 real" "screenshot $S/01-thread.png window" "logs")
