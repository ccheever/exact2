runs Electron.shell.showItemInFolder(bundle); Finder shows the bundle selected and comes to the front, and the helper

After (this branch, T3PermissionHelper.swift:72-87, 104):
    struct T3FinderReveal {
        static let finder = "com.apple.finder"
        var yield: (String) -> Void = { NSApp.yieldActivation(toApplicationWithBundleIdentifier: $0) }
        var select: (URL) -> Void = { NSWorkspace.shared.activateFileViewerSelecting([$0]) }
        var activate: (String) -> Void = { _ = NSRunningApplication.runningApplications(withBundleIdentifier: $0).first?.activate(options: []) }
        func reveal(_ url: URL) { yield(Self.finder); select(url); activate(Self.finder) }
    }
    var reveal: (URL) -> Void = { T3FinderReveal().reveal($0) }
  The helper's existing tracking then hides it: Settings not frontmost and the panel not key (sync(), unchanged).
