## 1. Compiler, three one-file inputs
--- state.contract
component App
  state family = "Arial"
  view
    text "Installed selected family" font-family=family
--- literal.contract
component App
  view
    text "Installed Arial" font-family="Arial, sans-serif"
--- choice.contract
component App
  state useArial = true
  view
    text "Installed selected family" font-family=(useArial ? "Arial" : "Menlo")
