# X68 record: an inspect layer cannot see what it covers
exact2 main b896050d78508aa5cea10cfa875e02037fde4c6e (2026-10-08); macOS 26.6.2 (25G83), Apple Silicon; web: Chrome 154.0.8037.98 (the agent's); agent mode only (no real input); every tap is a platform click, every hover a pointer move

## app.contract (bun scripts/exact.mjs new <dir>; app.ts has no sources)
component X68app
  state inspecting = false
  state layer = false
  state cancel = false
  state hit = "-"
  state presses = 0
  action arm(withLayer: bool)
    inspecting = true
    layer = withLayer
  action stop
    inspecting = false
    layer = false
  action toggleCancel
    cancel = not cancel
  action look(e: PointerEvent)
    if inspecting
      hit = match elementFromPoint(e.clientX, e.clientY) { case some(id) => id, case none => "none" }
  action pick(e: PointerEvent)
    if inspecting
      hit = match elementFromPoint(e.clientX, e.clientY) { case some(id) => id, case none => "none" }
      if cancel
        preventDefault()
  action pressPrimary
    presses = presses + 1
  view
    main testId="root" width="100%" height="100%" box-sizing="border-box" padding=24 gap=12 background-color="#ffffff" color="#111111" pointermove=look pointerdown=pick
      row gap=8
        button press=arm(false) testId="inspect"
          text "Inspect"
        button press=arm(true) testId="inspect-layer"
          text "Inspect (layer)"
        button press=stop testId="stop"
          text "Stop"
        button press=toggleCancel testId="cancel"
          text "preventDefault"
      text `hit: ${hit} · presses: ${presses}` white-space="nowrap" testId="out"
      text `inspecting: ${inspecting} · layer: ${layer} · preventDefault: ${cancel}` white-space="nowrap" testId="mode"
      column id="paint-surface-card" testId="card" width=280 padding=8 background-color="#eef2ff" border-width=1 border-style="solid" border-color="#c7d2fe" position="relative"
        button id="paint-primary-save" press=pressPrimary testId="primary" background-color="#2563eb" color="#ffffff" padding=8
          text "Save"
        when layer
          box testId="layer" position="absolute" left=0 top=0 width="100%" height="100%" pointermove=look pointerdown=pick

## Reading
1. No layer, hover over Save: hit = paint-primary-save on both hosts (elementFromPoint names the node; the id carries the role).
2. No layer, click Save: hit = paint-primary-save, but the app's own press also runs: presses 0 -> 1 on both hosts.
3. Same with preventDefault() in the root's pointerdown: presses 1 -> 2 on both hosts (a pointerdown cannot cancel the press, as in DOM).
4. Layer over the card, hover over Save: hit = paint-surface-card on both hosts (the layer is front-most; its nearest id is the card's), never paint-primary-save.
5. Layer, click over Save: web presses stay 2 (the layer takes the click); macOS presses 2 -> 3 (the click also pressed Save under the layer: the pass-through of #281, here with a plain positioned box).
