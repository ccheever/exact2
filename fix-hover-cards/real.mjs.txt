}

// #281's rule: a press inside a card, on a part with no press of its own, presses nothing under it. Find a point in the
// card's padding (or, failing that, anywhere in the card off its buttons) that lies over a control under the card.
function overControl(card, unders, pad, avoid = []) {
  const inside = (b, x, y) => x >= b.x && x < b.x + b.w && y >= b.y && y < b.y + b.h;
  let any = null;
  for (let y = card.y + 2; y < card.y + card.h - 2; y += 3) for (let x = card.x + 2; x < card.x + card.w - 2; x += 3) {
    if (avoid.some(b => inside(b, x, y)) || !unders.some(b => inside(b, x, y))) continue;
    const padding = x < card.x + pad.x || x > card.x + card.w - pad.x || y < card.y + pad.y || y > card.y + card.h - pad.y;
    if (padding) return { x, y, padding: true };
    any ??= { x, y, padding: false };
  }
  return any;
}
