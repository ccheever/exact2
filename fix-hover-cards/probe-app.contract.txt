// Probe: macOS hover semantics under a real pointer (fix-hover-cards).
// A: a hover box whose absolutely positioned child (the card) lies outside the box.
// B: a card with its own hover handler and a hover button inside it.
// C: a hover node under an absolutely positioned hover overlay (a later sibling).
shape Greeting
  text: string

component Probe
  resource greeting = greeting("Probe") as shape Greeting
  state log = ""
  state n = 0
  state aOver = false
  action ev(name: string, over: bool)
    n = n + 1
    log = `${n}${name}${over ? "+" : "-"} ${log}`
  action hA(over: bool)
    aOver = over
    ev("A", over)
  state nudge = 0
  state uOver = false
  action hU2(over: bool)
    uOver = over
    ev("V", over)
  action hD(over: bool)
    ev("D", over)
    if not over
      nudge = nudge + 1
  state eU = false
  state eC = false
  state fU = false
  state fC = false
  state fNudge = 0
  action hEU(over: bool)
    eU = over
  action hEC(over: bool)
    eC = over
  action hFU(over: bool)
    fU = over
  action hFC(over: bool)
    fC = over
    if not over
      fNudge = fNudge + 1
  action clear
    log = ""
    n = 0
  view
    main testId="root" width="100%" height="100%" box-sizing="border-box" padding=24 background-color="light-dark(#ffffff, #111111)" position="relative"
      row gap=12 align-items="center"
        button press=clear testId="clear" width=60 height=24
          text "clear"
        text greeting.text font-size=12 color="light-dark(#111111, #eeeeee)" testId="greeting"
      text `log: ${log}` font-size=13 line-height=18 color="light-dark(#111111, #eeeeee)" testId="log" max-width=840
      // A at 24,140 (window), 160x20; its card 160x80 below it.
      box position="absolute" left=24 top=140 width=160 height=20 background-color="#3b82f6" hover=hA testId="a"
        when aOver
          box position="absolute" left=0 top=20 width=160 height=80 background-color="#93c5fd" testId="a-card"
      // B: the mark 24x20 at 300,140, its card at 300,160 (160x80) with its own hover and a button 100x24 at 330,190.
      box position="absolute" left=300 top=140 width=160 height=20 background-color="#10b981" hover=ev("B") testId="b"
        box position="absolute" left=0 top=20 width=160 height=80 background-color="#6ee7b7" hover=ev("Bcard") testId="b-card"
          box position="absolute" left=30 top=30 width=100 height=24 background-color="#047857" hover=ev("Bbtn") testId="b-btn"
      // C: an under node 160x80 at 560,140, and a later sibling overlay 100x40 at 590,160 over it.
      box position="absolute" left=560 top=140 width=160 height=80 background-color="#f59e0b" hover=ev("U") testId="c-under"
      box position="absolute" left=590 top=160 width=100 height=40 background-color="#7c2d12" hover=ev("O") testId="c-over"
      // D: an under node 160x80 at 560,300 whose hover only recolours, and a later sibling card 100x40 at 590,320 over it that nudges its translate on a leave.
      box position="absolute" left=560 top=300 width=160 height=80 background-color=(uOver ? "#fde68a" : "#f59e0b") hover=hU2 testId="d-under"
      // E (no nudge) at 24,300 and F (nudge) at 300,300: an under node whose hover recolours it, a card over it whose hover recolours it.
      box position="absolute" left=24 top=300 width=160 height=80 background-color=(eU ? "#fde68a" : "#f59e0b") hover=hEU testId="e-under"
      box position="absolute" left=54 top=320 width=100 height=40 background-color=(eC ? "#22c55e" : "#1e3a8a") hover=hEC testId="e-card"
      box position="absolute" left=300 top=300 width=160 height=80 background-color=(fU ? "#fde68a" : "#f59e0b") hover=hFU testId="f-under"
      box position="absolute" left=330 top=320 width=100 height=40 background-color=(fC ? "#22c55e" : "#1e3a8a") translate=(fNudge % 2 == 0 ? "0px 0px" : "0.01px 0px") hover=hFC testId="f-card"
      box position="absolute" left=590 top=320 width=100 height=40 background-color="#1e3a8a" translate=(nudge % 2 == 0 ? "0px 0px" : "0.01px 0px") hover=hD testId="d-card"
