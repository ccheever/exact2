== Probe source (AFTER, target/devtools/after-probe/main.swift; the BEFORE probe is the same without the mermaid line)
import AppKit
import WebKit
// app-developer-tools AFTER probe: this branch's web views, built from the branch's module sources.
// Run as a bare executable (a development build) and from a bundle whose Resources carry
// distribution.json {"flavor":"packaged"} (the release line).
private let ignoreEvent: ExactNativeEventFn = { _, _, _, _, _ in }
_ = NSApplication.shared
NSApp.setActivationPolicy(.accessory)
print("bundle resources: \(Bundle.main.resourceURL?.path ?? "nil"), T3WebInspection.enabled=\(T3WebInspection.enabled)")
unsetenv("EXACT_ASSETS"); unsetenv("T3_TERMINAL_INSPECTABLE")
let plain = T3TerminalView(props: [:], events: ExactNativeEvents(fn: ignoreEvent, ctx: nil, nonce: 1), agent: true)
print("terminal (no EXACT_ASSETS) isInspectable=\(plain.web.isInspectable)")
plain.destroy()
setenv("EXACT_ASSETS", "/tmp", 1); setenv("T3_TERMINAL_INSPECTABLE", "1", 1)
let dev = T3TerminalView(props: [:], events: ExactNativeEvents(fn: ignoreEvent, ctx: nil, nonce: 1), agent: true)
print("terminal (EXACT_ASSETS and T3_TERMINAL_INSPECTABLE=1 set) isInspectable=\(dev.web.isInspectable)")
dev.destroy()
let media = R6MediaPreview(agent: true), host = NSView(frame: NSRect(x: 0, y: 0, width: 400, height: 300))
media.mount(host: host, kind: "html", raw: "http://127.0.0.1:9/api/assets/page?sig=1", name: "page.html")
print("html-preview isInspectable=\((host.subviews.first as? WKWebView)?.isInspectable.description ?? "no view")")
media.destroy()
print("mermaid isInspectable=\(T3TimelineMermaid.renderer().isInspectable)")

== Build and packaged copy (target/devtools/afterprobe.sh)
#!/bin/bash
set -u
cd "$(dirname "$0")/../.."
X=$(xcode-select -p); F="$X/Platforms/MacOSX.platform/Developer/Library/Frameworks"; L="$X/Platforms/MacOSX.platform/Developer/usr/lib"
A=target/devtools/after-probe; R="$PWD/target/t3-tests"
M=$(ls examples/t3-code/modules/apple/*.swift)
xcrun swiftc -swift-version 5 -module-name T3AfterProbe -F "$F" -I "$L" -L "$L" -Xlinker -rpath -Xlinker "$F" -Xlinker -rpath -Xlinker "$L" \
  host/apple/modules/ExactNativeModule.swift "$R/ExactDataKeys.swift" $M $A/main.swift -o $A/probe 2>&1 | grep -E "error:"
echo "== development (bare executable, no distribution.json)"
$A/probe 2>&1 | grep -E "isInspectable|enabled="
rm -rf $A/Packaged.app; mkdir -p $A/Packaged.app/Contents/MacOS $A/Packaged.app/Contents/Resources
cp $A/probe $A/Packaged.app/Contents/MacOS/probe
printf '{"flavor":"packaged"}' > $A/Packaged.app/Contents/Resources/distribution.json
/usr/libexec/PlistBuddy -c 'Add :CFBundleExecutable string probe' -c 'Add :CFBundleIdentifier string com.exact.t3code.inspection-probe' -c 'Add :CFBundlePackageType string APPL' $A/Packaged.app/Contents/Info.plist >/dev/null
echo "== packaged (Packaged.app/Contents/Resources/distribution.json {\"flavor\":\"packaged\"})"
$A/Packaged.app/Contents/MacOS/probe 2>&1 | grep -E "isInspectable|enabled="
