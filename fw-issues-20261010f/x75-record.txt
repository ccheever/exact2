// X75: two sends on one mutation in one turn. `bump` records each call when it
// starts and again when its work after an await finishes (app.ts). The field
// sends it on focus and on a key (the T3 clone's Files search); Escape sends it
// and blurs, and the blur sends it again.
shape Count
  text: string

component X75app
  resource counts = counts() as shape Count
  mutation bumped as shape Count
  state mode = "sync"
  state field = ""
  action pick(m: string)
    mode = m
  action edit(v: string)
    field = v
  action bump(tag: string)
    send bumped = bump(tag, mode)
  action onFocus
    bump("focus")
  action onBlur
    bump("blur")
  action onKey(k: string)
    bump(`key-${k}`)
    if k == "Escape"

== app.ts
import type { Answer, Result, Sources } from './app.contract.d.ts';

export const appId = 'com.example.x75app';
// A local server that answers after 300 ms (x75-server.mjs, run beside the drive).
export const grants = 'fs.write app:/data/x75.txt\nnet.fetch http://127.0.0.1:18775\n';

const started: string[] = [], finished: string[] = [], failed: string[] = [];
const encoder = new TextEncoder();

const sources: Sources = {
  counts: (): Result<'counts'> => ({ text: `started ${started.length} [${started.join(' ')}]\nfinished ${finished.length} [${finished.join(' ')}]\nfailed ${failed.length} [${failed.join(' ')}]` }),
  bump: async ([tag, mode], _store, storage): Promise<Result<'bump'>> => {
    const name = `${mode}:${tag}`;
    started.push(name);
    try {
      if (mode === 'store') await storage.fs.atomicWriteFile('app:/data/x75.txt', encoder.encode(name));
      else if (mode === 'fetch') await (await fetch(`http://127.0.0.1:18775/bump?tag=${encodeURIComponent(name)}`)).text();
      finished.push(name);
    } catch (error) {
      failed.push(`${name}(${(error as { kind?: string }).kind ?? String(error)})`);
      throw error;
    }
    return { text: name };
  },
};
export const answer: Answer = (source, args, store, storage, native) =>
  sources[source](args, store, storage, native);

== x75-server.mjs
// Answers /bump after 300 ms, with CORS for the web build; logs each request it receives.
Bun.serve({
  port: 18775, hostname: '127.0.0.1',
  async fetch(req) {
    const url = new URL(req.url);
    if (req.method === 'OPTIONS') return new Response(null, { headers: { 'access-control-allow-origin': '*', 'access-control-allow-headers': '*' } });
    console.log(`${new Date().toISOString()} received ${url.pathname}${url.search}`);
    await Bun.sleep(300);
    return new Response('ok', { headers: { 'access-control-allow-origin': '*' } });
  },
});
console.log('x75 server on 127.0.0.1:18775');

== Verdict: declared behaviour, not filed
sync: 4/4 finished on both hosts. store: 4/4 finished on both (the let-go calls' storage ran to its end natively, as the docs say).
fetch: web 4/4 finished; macOS 2/4, the superseded calls rejected with FetchError Aborted and never reached the server.
docs/reference.md:516-522 (main d413487a8): "A send dropped before it has issued storage does not run. A replaced answer waiting on a
`fetch` is not stranded: on the web build (the JS target) its fetch completes and the code after the `await` runs; natively and in the
web's wasm module realm the fetch rejects with a `FetchError` of kind `Aborted` (the request may already have been sent), so a `catch`
or `finally` runs. A stream's fetch never settles. Its reply is dropped either way; a mutation that needs every reply is declared `queue`."
docs/agent-pitfalls.md:327-334: "A superseded send's fetch rejects natively and completes on the web build. ... declare the mutation
`queue` when every send's reply matters (LLP 1092)." LLP 1016 D5, amended 2026-10-05 (llp/1016-async-data-settlement.rfc.md:257-265).
native.later is a fetch to exact-native: (js/src/prelude.js:1137), so the T3 clone's native.later call falls under the same rule.
