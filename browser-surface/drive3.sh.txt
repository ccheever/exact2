#!/bin/zsh
# browser-surface part 1: the one live drive (agent mode, lane env). Not committed.
W=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-browser-surface
L=$W/target/browser-surface/lane
P=${1:-drive}; OUT=$L/runs/$P; rm -rf $OUT; mkdir -p $OUT
BUN=$HOME/.bun-1.4.2/bin/bun
export EXACT_APP_DIR=$W/examples/t3-code
export T3_LOCAL_HOME=$L/t3-home T3_LOCAL_PORT=16650 T3CODE_TELEMETRY_ENABLED=false
export CODEX_HOME=$L/codex CLAUDE_CONFIG_DIR=$L/claude
export XDG_CONFIG_HOME=$L/xdg/config XDG_DATA_HOME=$L/xdg/data XDG_STATE_HOME=$L/xdg/state XDG_CACHE_HOME=$L/xdg/cache
export T3_REMOTE_OPEN_LOG=$OUT/opened-urls.txt
export PATH=$L/bin:$HOME/.bun-1.4.2/bin:/usr/bin:/bin:/usr/sbin:/sbin
export T3_LOCAL_RUNTIME_DIR=$L/runtime/t3-0.0.46-nightly.20261005.2667-darwin-arm64
# A seeded lane home: two projects (`t3 project add`), so the first-run gate opens the app, not the wizard.
rm -rf $T3_LOCAL_HOME; cp -R $L/t3-home-seed $T3_LOCAL_HOME
# The lane fixture server (lane-served pages) on 16651.
FIXTURE_PORT=16651 $BUN $W/target/browser-surface/fixture.mjs > $OUT/fixture.log 2>&1 &
FX=$!; echo "fixture pid $FX" > $OUT/pids.txt
waitline() { while [ $(cat $OUT/agent.jsonl 2>/dev/null | wc -l) -lt $1 ]; do sleep 0.2; done; }
cd $W
S=$OUT
$BUN scripts/agent.mjs macos --json --size 1280x840 \
  "clock +15000 real" "clock +6000 real" \
  "tap toggle-right-panel" "clock +1500 real" \
  "tap surface-browser" "clock +3000 real" \
  "type browser-url 127.0.0.1:16651" "type browser-url key Enter" "clock +4000 real" "screenshot $S/03-lane-page.png" logs \
  "type browser-url 127.0.0.1:16651/a" "type browser-url key Enter" "clock +3000 real" \
  "type browser-url 127.0.0.1:16651/b" "type browser-url key Enter" "clock +3000 real" "screenshot $S/04-page-b.png" \
  "tap browser-back" "clock +3000 real" "screenshot $S/05-back.png" tree \
  "tap browser-forward" "clock +3000 real" "screenshot $S/06-forward.png" \
  "type browser-url 127.0.0.1:16651/slow?ms=9000" "type browser-url key Enter" "clock +2000 real" "screenshot $S/07-loading.png" tree \
  "tap browser-refresh" "clock +1500 real" "screenshot $S/08-stop-pressed.png" "clock +11000 real" "screenshot $S/09-slow-loaded.png" \
  "type browser-url example.com" "type browser-url key Enter" "clock +7000 real" "screenshot $S/10-public.png" \
  "tap browser-address hover" "clock +300 real" "tap browser-open-external" "clock +500 real" \
  "type browser-url 127.0.0.1:16699" "type browser-url key Enter" "clock +4000 real" "screenshot $S/11-failed.png" tree \
  "tap browser-failed-reload" "clock +3000 real" \
  "type browser-url 127.0.0.1:16651/a" "type browser-url key Enter" "clock +3000 real" \
  "tap browser-more" "clock +1000 real" "screenshot $S/12-more.png" "tap browser-more-hard-reload" "clock +3000 real" \
  "tap panel-toggle-right" "clock +1500 real" "screenshot $S/13-hidden.png" \
  "tap toggle-right-panel" "clock +2500 real" "screenshot $S/14-reshown.png" tree logs \
  "tap close-tab-browser:tab_1" "clock +2500 real" "screenshot $S/15-closed.png" \
  logs state > $OUT/agent.jsonl 2> $OUT/agent.err &
AG=$!; echo "agent pid $AG" >> $OUT/pids.txt
waitline 1
# The lane project: `t3 app` opens it in the running app (desktop-activation).
env -i HOME=$L/home USER=$USER PATH=/usr/bin:/bin TMPDIR="$TMPDIR" T3CODE_TELEMETRY_ENABLED=false $L/runtime/t3-0.0.46-nightly.20261005.2667-darwin-arm64/t3 app $L/projects/work --base-dir $T3_LOCAL_HOME > $OUT/t3-app.out 2>&1; echo "t3 app exit $?" >> $OUT/pids.txt
echo "server pid $(ps -axo pid,command | grep -v grep | grep -- "--bootstrap-fd" | grep "$L" | awk '{print $1}' | head -1)" >> $OUT/pids.txt
wait $AG; echo "agent exit $?" >> $OUT/pids.txt
kill $FX
echo DRIVE-DONE
