#!/bin/zsh
# browser-surface: a normal launch of a lane copy for real input (not committed).
W=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-browser-surface
L=$W/target/browser-surface/lane
OUT=$L/runs/${1:-real}; rm -rf $OUT; mkdir -p $OUT
SRC="$W/target/clients/6eeec44af8ed4603e489acb0/com.exact.t3code.macos/macos/T3 Code (Exact).app"
APP="$L/apps/T3 Code (Lane Browser).app"
rm -rf "$APP"; cp -R "$SRC" "$APP"
PB=/usr/libexec/PlistBuddy
$PB -c "Set :CFBundleIdentifier com.exact.t3code.macos.lanebrowser" "$APP/Contents/Info.plist"
$PB -c "Set :CFBundleName T3 Code (Lane Browser)" "$APP/Contents/Info.plist"
$PB -c "Delete :CFBundleDisplayName" "$APP/Contents/Info.plist" 2>/dev/null
$PB -c "Add :CFBundleDisplayName string T3 Code (Lane Browser)" "$APP/Contents/Info.plist"
codesign --force --deep -s - "$APP" > $OUT/codesign.log 2>&1 || { echo codesign failed; exit 1; }
BUN=$HOME/.bun-1.4.2/bin/bun
rm -rf $L/t3-home; cp -R $L/t3-home-seed $L/t3-home
mkdir -p $L/home
FIXTURE_PORT=16651 $BUN $W/target/browser-surface/fixture.mjs > $OUT/fixture.log 2>&1 &
echo "fixture $!" > $OUT/pids.txt
env -i HOME=$L/home CFFIXED_USER_HOME=$L/home USER=$USER LOGNAME=$USER TMPDIR="$TMPDIR" LANG=en_US.UTF-8 \
  PATH=$L/bin:/usr/bin:/bin:/usr/sbin:/sbin \
  T3_LOCAL_HOME=$L/t3-home T3_LOCAL_PORT=16650 T3CODE_TELEMETRY_ENABLED=false \
  T3_LOCAL_RUNTIME_DIR=$L/runtime/t3-0.0.46-nightly.20261005.2667-darwin-arm64 \
  CODEX_HOME=$L/codex CLAUDE_CONFIG_DIR=$L/claude \
  XDG_CONFIG_HOME=$L/xdg/config XDG_DATA_HOME=$L/xdg/data XDG_STATE_HOME=$L/xdg/state XDG_CACHE_HOME=$L/xdg/cache \
  T3_REMOTE_OPEN_LOG=$OUT/opened-urls.txt \
  "$APP/Contents/MacOS/T3 Code (Exact)" > $OUT/app.log 2>&1 &
echo "app $!" >> $OUT/pids.txt
echo "out $OUT"
