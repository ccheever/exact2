// RE-1 (realinput-1010e-followups): the reference's permission helper window (MacPermissionHelper.ts:100-117, the same
// BrowserWindow options, shown with showInactive) on the reference's own Electron 44.4.2, and AppKit's answer to whether
// the view under its "T3 Code" row takes the click that activates the app. `--hold <ms>` keeps the window up for a real
// click (the real-input step): the page then logs every mousedown and click it receives.
const { app, BrowserWindow, nativeImage, screen } = require("electron");
const path = require("node:path");
app.setPath("userData", path.join(__dirname, "user-data"));
app.dock?.hide();
const hold = Number(process.argv[process.argv.indexOf("--hold") + 1]) || 0;
const html = `<!doctype html><html><head><meta charset="utf-8"><title>Set up Screen Recording</title><style>
* { box-sizing: border-box; } html { background: transparent; } body { margin: 0; background: transparent; font: 15px/22px -apple-system, sans-serif; user-select: none; }
#panel { position: relative; margin: 2px; padding: 20px; height: 136px; border: 1px solid #e3e3e3; border-radius: 24px; background: #fff; }
#close { position: absolute; right: 8px; top: 6px; width: 22px; height: 22px; padding: 0; border: 0; border-radius: 50%; background: #fff; font-size: 18px; }
#app { display: flex; align-items: center; gap: 12px; width: 100%; height: 52px; margin-top: 20px; padding: 8px 12px; border: 0; border-radius: 10px; background: #e7e7e7; cursor: grab; font-size: 16px; font-weight: 600; }
</style></head><body><main id="panel"><button id="close" aria-label="Close permission helper">×</button><header>↑ Drag T3 Code into the list above</header>
<button id="app" draggable="true">T3 Code</button></main><script>
for (const type of ["mousedown", "click", "dragstart"]) document.addEventListener(type, (event) => console.log("page " + type + " on #" + (event.target.id || event.target.tagName)), true);
document.getElementById("app").addEventListener("dragstart", (event) => { event.preventDefault(); console.log("page drag-request"); });
</script></body></html>`;
const variants = [["reference: alwaysOnTop, no acceptFirstMouse (MacPermissionHelper.ts:100-117)", {}], ["control: the same window at the normal level (alwaysOnTop false)", { alwaysOnTop: false }]];
for (const name of ["did-become-active", "did-resign-active"]) app.on(name, () => console.log(`${new Date().toISOString()} app ${name}`));
app.whenReady().then(async () => {
  const probe = { exports: {} };
  process.dlopen(probe, path.join(__dirname, "probe.node"));
  const area = screen.getPrimaryDisplay().workArea;
  const pick = Number(process.argv[process.argv.indexOf("--variant") + 1]) || 0;
  for (const [label, extra] of [variants[pick]]) {
    const window = new BrowserWindow({ width: 560, height: 140, show: false, frame: false, transparent: true, roundedCorners: false, backgroundColor: "#00000000",
      hasShadow: false, resizable: false, minimizable: false, maximizable: false, fullscreenable: false, alwaysOnTop: true, skipTaskbar: true,
      title: "Set up Screen Recording", ...extra, webPreferences: { sandbox: true, contextIsolation: true, nodeIntegration: false, partition: `probe-${Date.now()}` } });
    window.webContents.on("console-message", (event) => {
      console.log(`${new Date().toISOString()} ${event.message}`);
      // The reference's "drag" message (MacPermissionHelper.ts:141-146): a native drag of a file with the 64 pt icon.
      if (event.message === "page drag-request") {
        const icon = nativeImage.createFromPath(path.join(__dirname, "icon.png")).resize({ width: 64, height: 64 });
        window.webContents.startDrag({ file: path.join(__dirname, "package.json"), icon });
        console.log(`${new Date().toISOString()} startDrag returned`);
      }
    });
    for (const name of ["focus", "blur"]) window.on(name, () => console.log(`${new Date().toISOString()} window ${name}`));
    await window.loadURL(`data:text/html;charset=utf-8,${encodeURIComponent(html)}`);
    window.setBounds({ x: area.x + area.width - 580, y: area.y + area.height - 160, width: 560, height: 140 }, false);
    window.showInactive();
    await new Promise((done) => setTimeout(done, 600));
    const rect = await window.webContents.executeJavaScript(`JSON.stringify(Object.fromEntries(["app", "close"].map((id) => { const r = document.getElementById(id).getBoundingClientRect(); return [id, { x: r.x + r.width / 2, y: r.y + r.height / 2 }]; })))`);
    const centers = JSON.parse(rect);
    const answers = Object.fromEntries(Object.entries(centers).map(([id, point]) => [id, JSON.parse(probe.exports.probe(window.getTitle(), point.x, point.y))[0]]));
    console.log(JSON.stringify({ variant: label, electron: process.versions.electron, alwaysOnTop: window.isAlwaysOnTop(), focused: window.isFocused(), centers, answers }));
    if (hold) { console.log(`holding ${hold} ms for a real click (the page logs what reaches it)`); await new Promise((done) => setTimeout(done, hold)); }
    window.destroy();
  }
  app.quit();
});
