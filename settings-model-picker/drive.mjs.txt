async function waitFor(what, pred, ms = 30000) {
  const t0 = Date.now();
  for (;;) {
    let ok = false; try { ok = await pred(); } catch { ok = false; }
    if (ok) { log('ready', { what, ms: Date.now() - t0 }); return true; }
    if (Date.now() - t0 > ms) { log('timeout', { what, ms, ids: [...await ids()].slice(0, 200) }); return false; }
    await real(500);
  }
}

function pick(st, name) {
  for (const group of ['resources', 'slots', 'derives']) if (st?.[group] && name in st[group]) return st[group][name];
  return st?.[name];
}

}
