#!/bin/zsh
# files-gutter-parity: one agent drive of the clone. Usage: drive.sh before|after
# Clone server on 16102: the development build takes lane ports 16000-16999 only (T3LocalBackend laneRange), so not 17102.
# before: the feature tip eb9752918 built in t3-code-evidence-base (never edited); after: this worktree's bundle. Same steps, 1280x840.
MODE=$1
A=/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-audit
W=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-files-gutter-parity
S=$W/target/fgp/drive/$MODE; mkdir -p $S
if [ "$MODE" = before ]; then ROOTDIR=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-evidence-base; LANE=files-gutter-parity-before; else ROOTDIR=$W; LANE=files-gutter-parity; fi
F=.agents/skills/verify/SKILL.md
ops=(
  "clock +20000 real" "clock +3000 real"
  "tap thread-verify-timeline" "clock +2500 real"
  "tap details-changes" "clock +10000 real"
  'tap "Dismiss notification"' "clock +1500 real"
  "tap \"Expand $F\"" "clock +4000 real"
  # FG-4: a skill chip's details beside the Diff panel, then a press on the gutter's number 2.
  'type composer $frontend-design now' "clock +1500 real"
  "tap composer mouse at 60 12" "clock +1000 real"
  "screenshot $S/4a-chip-open.png window"
  "tap diff-line-$F-additions-2" "clock +1500 real"
  "screenshot $S/4b-diff-gutter-press.png window"
  "tap diff-line-$F-additions-2" "clock +1000 real"
  # FG-3: split view, light.
  "tap diff-split" "clock +2000 real"
  "screenshot $S/3-split-light.png window"
  "tap diff-stacked" "clock +1500 real"
  # FG-1, FG-2: the Files preview, light.
  "tap panel-add-surface" "clock +1500 real"
  "tap add-surface-files" "clock +4000 real"
  "tap tree-fixture.txt" "clock +4000 real"
  "tap file-line-1 hover" "clock +1000 real"
  "screenshot $S/1-files-hover-light.png window"
  "tap file-line-1" "clock +1500 real"
  "screenshot $S/2c-files-click-light.png window"
  "tap diff-comment-cancel" "clock +1000 real"
  "tap file-line-1 drag to file-line-2 mouse over 400 during \"clock +1500 real\" \"screenshot $S/2a-files-drag-held-light.png window\""
  "clock +1500 real"
  "screenshot $S/2b-files-drag-released-light.png window"
  "tap file-line-1" "clock +1000 real"
  "tap diff-comment-cancel" "clock +1000 real"
  # FG-4 on the Files gutter (the gutter now hears pointerdown): the chip's details close.
  "tap composer mouse at 60 12" "clock +1000 real"
  "screenshot $S/4c-chip-open-files.png window"
  "tap file-line-1" "clock +1500 real"
  "screenshot $S/4d-files-gutter-press.png window"
  "tap diff-comment-cancel" "clock +1000 real"
  # Dark: the Files tab opened again in dark (a live switch leaves some texts light, X-scheme), then split view.
  "prefer prefers-color-scheme dark" "clock +1500 real"
  "tap close-tab-file:fixture.txt" "clock +1500 real"
  "tap panel-add-surface" "clock +1500 real"
  "tap add-surface-files" "clock +4000 real"
  "tap tree-fixture.txt" "clock +4000 real"
  "tap file-line-1 hover" "clock +1000 real"
  "screenshot $S/1-files-hover-dark.png window"
  "tap file-line-1 drag to file-line-2 mouse over 400 during \"clock +1500 real\" \"screenshot $S/2a-files-drag-held-dark.png window\""
  "clock +1500 real"
  "screenshot $S/2b-files-drag-released-dark.png window"
  "tap file-line-1" "clock +1000 real"
  "tap diff-comment-cancel" "clock +1000 real"
  "tap tab-diff" "clock +2000 real"
  "tap diff-split" "clock +2500 real"
  "screenshot $S/3-split-dark.png window"
  "tap diff-stacked" "clock +1000 real"
  "prefer prefers-color-scheme light" "clock +1000 real"
)
print -rl -- "${ops[@]}" > $S/ops.txt
EXACT_ROOT=$ROOTDIR CLONE_STORAGE=audit-fgp2-$MODE $A/clone-drive.sh $LANE 16100 "${ops[@]}" > $S/drive.jsonl 2> $S/drive.err
echo "exit $?"
