// files-gutter-parity FG-1/FG-2: the reference's Files surface (FilePreviewPanel's File over @pierre/diffs), fixture.txt.
// Reads the number cells, the "+" ([data-utility-button]) on a hovered line, light and dark; then a click on number 1 and a
// drag over numbers 1 -> 2: the selection and whether the draft is open while held and after the release.
// Usage: node ref-fg12.mjs <cdpPort> <shotPrefix>
import { open, log } from './ref-lib.mjs';
const [port, shot] = process.argv.slice(2);
const page = await open(port);
const tree = page.getByRole('treeitem', { name: 'fixture.txt' });
if (await tree.count()) { await tree.first().click(); await page.waitForTimeout(3000); }
const read = () => page.evaluate(() => {
  const cv = document.createElement('canvas'); cv.width = cv.height = 1; const cx = cv.getContext('2d', { willReadFrequently: true });
  const hex = (c) => { if (!c || c === 'rgba(0, 0, 0, 0)') return c; cx.clearRect(0, 0, 1, 1); cx.fillStyle = '#000'; cx.fillStyle = c; cx.fillRect(0, 0, 1, 1); const d = cx.getImageData(0, 0, 1, 1).data; return '#' + [d[0], d[1], d[2]].map(v => v.toString(16).padStart(2, '0')).join('') + (d[3] < 255 ? d[3].toString(16).padStart(2, '0') : ''); };
  const box = (el) => { const r = el.getBoundingClientRect(); return { x: +r.x.toFixed(1), y: +r.y.toFixed(1), w: +r.width.toFixed(1), h: +r.height.toFixed(1) }; };
  const out = { rows: [], plus: [], drafts: [], selection: getSelection().toString() };
  for (const c of document.querySelectorAll('diffs-container')) {
    const root = c.shadowRoot; if (!root) continue;
    for (const el of root.querySelectorAll('[data-column-number]')) {
      const r = el.getBoundingClientRect(); if (r.width === 0) continue;
      const s = getComputedStyle(el);
      out.rows.push({ n: el.getAttribute('data-column-number'), sel: el.getAttribute('data-selected-line'), ...box(el), bg: hex(s.backgroundColor), color: hex(s.color), pl: s.paddingLeft, pr: s.paddingRight, us: s.userSelect, cursor: s.cursor });
    }
    for (const b of root.querySelectorAll('[data-utility-button]')) {
      const s = getComputedStyle(b); const r = b.getBoundingClientRect(); if (r.width === 0) continue;
      const svg = b.querySelector('svg'); const sv = svg ? getComputedStyle(svg) : null;
      out.plus.push({ ...box(b), bg: hex(s.backgroundColor), color: hex(s.color), radius: s.borderRadius, opacity: s.opacity, visibility: s.visibility, svg: svg ? { ...box(svg), color: hex(sv.color), fill: sv.fill } : null, parent: b.parentElement?.getAttribute('data-column-number') ?? b.parentElement?.tagName });
    }
  }
  for (const t of document.querySelectorAll('textarea')) { const r = t.getBoundingClientRect(); if (r.width > 0) out.drafts.push({ ph: t.placeholder, ...box(t) }); }
  const label = [...document.querySelectorAll('*')].filter(e => e.children.length === 0 && /^Comment on line|^Line|^Lines/.test((e.textContent || '').trim())).map(e => e.textContent.trim());
  out.labels = label.slice(0, 5);
  return out;
});
const cancel = async () => { const b = page.getByRole('button', { name: 'Cancel', exact: true }); if (await b.count()) { await b.first().click(); await page.waitForTimeout(600); } };
let g = (await read()).rows;
log({ phase: 'open', rows: g });
const one = g.find(e => e.n === '1'), two = g.find(e => e.n === '2');
// hover line 1's code: the "+" shows
await page.mouse.move(one.x + one.w + 60, one.y + 10); await page.waitForTimeout(600);
for (const scheme of ['light', 'dark']) { await page.emulateMedia({ colorScheme: scheme }); await page.waitForTimeout(900); log({ phase: 'hover', scheme, ...(await read()) }); await page.screenshot({ path: `${shot}-hover-${scheme}.png` }); }
await page.emulateMedia({ colorScheme: 'light' }); await page.waitForTimeout(600);
// a click in the middle of number 1
await page.mouse.move(one.x + one.w / 2, one.y + 10); await page.waitForTimeout(300);
await page.mouse.down(); await page.waitForTimeout(250); log({ phase: 'click-held', ...(await read()) });
await page.mouse.up(); await page.waitForTimeout(900); log({ phase: 'click-released', ...(await read()) }); await page.screenshot({ path: `${shot}-click.png` });
await cancel(); log({ phase: 'cancelled', ...(await read()) });
// a drag over the numbers 1 -> 2
await page.mouse.move(one.x + one.w - 8, one.y + 10); await page.waitForTimeout(300); await page.mouse.down();
for (let i = 1; i <= 6; i++) { await page.mouse.move(one.x + one.w - 8, one.y + 10 + (two.y - one.y) * i / 6); await page.waitForTimeout(40); }
await page.waitForTimeout(400); log({ phase: 'drag-held', ...(await read()) }); await page.screenshot({ path: `${shot}-drag-held.png` });
await page.mouse.up(); await page.waitForTimeout(900); log({ phase: 'drag-released', ...(await read()) }); await page.screenshot({ path: `${shot}-drag-released.png` });
await cancel();
// a press on the "+" of line 2 (hovered) and a drag up to 1
await page.mouse.move(two.x + two.w + 60, two.y + 10); await page.waitForTimeout(500);
const plus = (await read()).plus[0]; log({ phase: 'plus-on-2', plus });
if (plus) {
  await page.mouse.move(plus.x + plus.w / 2, plus.y + plus.h / 2); await page.waitForTimeout(300); await page.mouse.down(); await page.waitForTimeout(200);
  log({ phase: 'plus-held', ...(await read()) });
  await page.mouse.up(); await page.waitForTimeout(900); log({ phase: 'plus-released', ...(await read()) }); await page.screenshot({ path: `${shot}-plus.png` });
  await cancel();
}
process.exit(0);

log({ at });
const numbers = () => page.evaluate(() => { let n = 0; for (const c of document.querySelectorAll('diffs-container')) { const root = c.shadowRoot; if (root) n += root.querySelectorAll('[data-column-number]').length; } return n; });

  const cv = document.createElement('canvas'); cv.width = cv.height = 1; const cx = cv.getContext('2d', { willReadFrequently: true });

  const out = [];
  for (const c of document.querySelectorAll('diffs-container')) {
