  // ── Dark, reduced motion, 840×620 ─────────────────────────────────────────
  await step('looks', async () => {
    await s.prefer({ 'prefers-color-scheme': 'dark' }); await settle(1500);
    await reveal('docs/usage.md'); await shot('26-dark');

    await s.resize(840, 620); await settle(1500); await shot('27-840x620');
    await s.resize(1280, 840); await settle(1000);
  });

  // ── #115: Load more, the conversations off the diff (reduced motion) ──────
  await step('reviewed-pr', async () => {
    await s.tap('pull-request-tab-summary'); await settle(500);
    await openPr(REVIEWED); await settle(2000);
    await s.tap('pull-request-tab-code'); await settle(1000);
    await waitFor(async () => /file/.test(await textOf('pull-request-code-files')), 60_000);
    await s.tap('pull-request-code-fold-src/validate.js'); await settle(1500);
    await shot('28-115-code');
    rec(`orphans: "${await prop('pull-request-code-orphans-toggle', 'aria-label')}"`);
    const more = (await idsIn('pull-request-code-list')).find((id) => id.startsWith('pull-request-thread-more-'));
    rec(`load more button: ${more}`);
    if (more) {
      const before = (await threadIds()).length;
      await s.tap(more); await settle(1000);
      await waitFor(async () => (await threadIds()).length > before, 30_000);
      rec(`comments shown before ${before}, after ${(await threadIds()).length}`);
      await shot('29-115-load-more');
    }
    await s.prefer({ 'prefers-reduced-motion': 'reduce' }); await settle(500);
    await s.tap('pull-request-code-orphans-toggle'); await settle(1000);
    rec(`orphans open: ${await prop('pull-request-code-orphans-toggle', 'aria-expanded')}`);
    await shot('30-115-orphans-reduced-motion');
    await dump('30-orphans', 'pull-request-code-orphans');
    await s.prefer({ 'prefers-reduced-motion': 'no-preference' }); await settle(300);
  });

  // ── Add to agent (a new thread with the lines and the request) ────────────
  await step('add-to-agent', async () => {
    await s.tap('diff-comment-src/validate.js-additions-4'); await settle(500);
    await type('pull-request-comment-text', 'Explain why an empty input is an error here.');

    rec(`toasts: ${await toastsNow()}`);
    await settle(3000); await shot('32-agent-thread');
    await dump('32-composer', 'composer');
  });
  // ── Fix in a thread (the shared hand-off) from a conversation on #115 ──────
  await step('fix-in-a-thread', async () => {
    await s.tap('open-pull-requests'); await settle(3000); await openPr(REVIEWED);
    await s.tap('pull-request-tab-code'); await settle(3000);
    const fix = (await idsIn('pull-request-code-panel')).find((id) => id.startsWith('pull-request-thread-fix-'));
    rec(`fix button: ${fix}`);
    if (fix) { await tapIn('pull-request-code-panel', fix); await settle(5000); rec(`toasts: ${await toastsNow()}`); await shot('33-fix'); }
  });
} finally {
  rec('closing');
  try { await s.close(); } catch {}
}

async function pushAsSecond(path, text) {
  const dir = resolve(MINE, 'target/drive/push-clone');
  const git = (args, opts = {}) => lib.git(paths, 'second', args, opts);
  const pr = gh('primary', [`repos/${REPO}/pulls/${CODE}`]).body;
  try { git(['clone', '--quiet', '--branch', pr.head.ref, `https://github.com/${REPO}.git`, dir]); } catch { git(['fetch', '--quiet', 'origin'], { cwd: dir }); git(['checkout', '--quiet', '-B', pr.head.ref, `origin/${pr.head.ref}`], { cwd: dir }); }
  writeFileSync(resolve(dir, path), text);
  git(['add', path], { cwd: dir });
  git(['-c', 'user.name=daehyeon-mun', '-c', 'user.email=<redacted noreply address>', 'commit', '--quiet', '-m', 'Revise the first entry'], { cwd: dir });
  git(['push', '--quiet', 'origin', `HEAD:${pr.head.ref}`], { cwd: dir });
  return git(['rev-parse', 'HEAD'], { cwd: dir }).stdout.trim();
}
