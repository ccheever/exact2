#!/bin/zsh
# One agent drive of the clone for diff-gutter-visuals. Usage: drive.sh before|after
# before: the feature tip ca034e055 built in t3-code-diff-gutter-visuals-before; after: this worktree's bundle.
# Same steps, same size (1280x840). (1) The fixture thread's Diff panel (Changes), .agents/skills/verify/SKILL.md (+5): numbers 2 to 4
# dragged and released (they stay selected; the "+" on 4), light then dark. (2) The GitHub lane's server paired
# (`bun lane.mjs start primary --port 16743`, a fresh single-use link), #168's Code tab, docs/usage.md: new line 3's
# number dragged to added line 5's and held for the screenshot (the release opens a draft, then Cancel), light then dark.
MODE=$1
A=/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-audit
W=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-diff-gutter-visuals
S=$W/target/dgv-drive/shots/$MODE; mkdir -p $S
if [ "$MODE" = before ]; then ROOTDIR=/Users/daehyeonmun/orca/workspaces/exact2/t3-code-diff-gutter-visuals-before; LANE=diff-gutter-visuals-before; else ROOTDIR=$W; LANE=diff-gutter-visuals; fi
G=$W/examples/t3-code/tools/github-lane
( cd $G && PATH=$HOME/.bun-1.4.2/bin:$PATH T3_GITHUB_LANE_SHARED=/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-ui-parity/github-lane bun lane.mjs pair primary >/dev/null )
PAIR=$(cat $W/target/github-lane/servers/primary/pairing-url); rm -f $W/target/github-lane/servers/primary/pairing-url
D=.agents/skills/verify/SKILL.md
# Reads of the after build's pinned "+" (the before build's press on the number lands on its hidden "+", which opens a
# draft and hides it, so it has no "+" there to read).
READS=("tree diff-line-$D-additions-4")
[ "$MODE" = after ] && READS+=("tree diff-comment-$D-additions-4")
F=docs/usage.md
ops=(
  "clock +20000 real" "clock +3000 real"
  # (1) The thread's Diff panel.
  "tap thread-verify-timeline" "clock +2500 real"
  "tap details-changes" "clock +10000 real"
  "tap \"Expand $D\"" "clock +4000 real"
  "screenshot $S/1-diff-open.png window"
  "tap diff-line-$D-additions-2 drag to diff-line-$D-additions-4 mouse over 400" "clock +1500 real"
  "screenshot $S/2-diff-light.png window"
  "${READS[@]}"
  "prefer prefers-color-scheme dark" "clock +1500 real"
  "screenshot $S/3-diff-dark.png window"
  "prefer prefers-color-scheme light" "clock +1000 real"
  # (2) Pair the GitHub lane's server (the embedded one stays focused), then #168's Code tab.
  "tap connection-settings" "clock +1500 real"
  "tap settings-connections" "clock +1500 real"
  "tap manage-connection" "clock +1000 real"
  "type server-origin $PAIR" "clock +500 real"
  "tap connect" "clock +12000 real"
  "type t3-code key Escape" "clock +1500 real"
  "tap open-pull-requests" "clock +6000 real"
  "type pull-requests-search #168" "clock +6000 real"
  "tap pr-row-168" "clock +6000 real"
  "tap pull-request-tab-code" "clock +15000 real"
  "tap pull-request-code-tree-toggle" "clock +1500 real"
  "tap pull-request-code-tree-more" "clock +6000 real"
  "tap pull-request-code-tree-more" "clock +6000 real"
  "tap pull-request-code-tree-more" "clock +6000 real"
  "tap pull-request-code-tree-toggle" "clock +1500 real"
  "tap pull-request-code-list into file:$F block start" "clock +1500 real"
  "tap pull-request-code-fold-$F" "clock +4000 real"
  "tap pull-request-code-list into file:$F block start" "clock +1500 real"
  "screenshot $S/4-usage.png window"
  "tap diff-line-$F-additions-3 drag to diff-line-$F-additions-5 mouse over 400 during \"clock +1500 real\" \"screenshot $S/5-pr-light.png window\""
  "clock +2000 real"
  "tap pull-request-comment-cancel" "clock +1500 real"
  "prefer prefers-color-scheme dark" "clock +1500 real"
  "tap pull-request-code-list into file:$F block start" "clock +1500 real"
  "tap diff-line-$F-additions-3 drag to diff-line-$F-additions-5 mouse over 400 during \"clock +1500 real\" \"screenshot $S/6-pr-dark.png window\""
  "clock +2000 real"
  "tap pull-request-comment-cancel" "clock +1500 real"
  "prefer prefers-color-scheme light" "clock +500 real"
  "state"
  "logs"
)
print -rl -- "${ops[@]//$PAIR/<pairing link>}" > $S/ops.txt
EXACT_ROOT=$ROOTDIR $A/clone-drive.sh $LANE 16740 --epoch 2026-10-11T09:00:00Z --time-zone Asia/Seoul "${ops[@]}" > $S/drive.jsonl 2> $S/drive.err
echo "exit $?"
