import os

OUT = os.path.join(U, 'evidence'); os.makedirs(OUT, exist_ok=True)
try:
    FONT = ImageFont.truetype('/System/Library/Fonts/SFNS.ttf', 22)
    SMALL = ImageFont.truetype('/System/Library/Fonts/SFNS.ttf', 18)
except Exception:
    FONT = SMALL = ImageFont.load_default()

def panel(path, crop=None, scale=2, height=None):
    """A clone screenshot (2x) cropped by a CSS-pixel box, or a reference shot (1x) as is; scaled to `height`."""
    im = Image.open(path).convert('RGB')
    if crop:
        x0, y0, x1, y1 = crop
        im = im.crop((x0 * scale, y0 * scale, x1 * scale, y1 * scale))
    if height:
        im = im.resize((round(im.width * height / im.height), height), Image.LANCZOS)
    return im

def compose(name, caption, items, height):
    tiles = [(label, panel(path, crop, scale, height)) for label, path, crop, scale in items]
    pad, head, cap = 16, 34, 40
    width = sum(t.width for _, t in tiles) + pad * (len(tiles) + 1)
    out = Image.new('RGB', (width, cap + head + height + pad), 'white')
    d = ImageDraw.Draw(out)
    d.text((pad, 10), caption, fill='#222', font=SMALL)
    x = pad
    for label, tile in tiles:
        d.text((x, cap + 4), label, fill='#c2185b' if label.startswith('before') else '#1b5e20' if label.startswith('after') else '#0d47a1', font=FONT)
        out.paste(tile, (x, cap + head))
        d.rectangle((x - 1, cap + head - 1, x + tile.width, cap + head + tile.height), outline='#bbb')
        x += tile.width + pad
    out.save(os.path.join(OUT, name))
    print(name, out.size)
