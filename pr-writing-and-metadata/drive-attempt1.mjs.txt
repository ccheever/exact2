// pr-writing-and-metadata live drive (not committed): bun drive.mjs <repo> <label> <after|before>
// One macOS agent session against the primary lane server (127.0.0.1:16760, real GitHub through the
// lane gh). Pairs with a fresh single-use token (never printed). `after` drives every write on the
// task's own pull requests (#158 by the second account, #159 by the primary) and #115's seeded
// comment with two reactions, reads each effect back from GitHub with the lane gh, and undoes what
// can be undone. `before` takes the same screens on the base build (no writes).
import { appendFileSync, mkdirSync, readFileSync, statSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
const [repo, label, mode = 'after'] = process.argv.slice(2);
const MINE = '/Users/daehyeonmun/orca/workspaces/exact2/t3-code-pr-writing-and-metadata';
const L = resolve(import.meta.dir), out = resolve(L, `out-${label}`); mkdirSync(out, { recursive: true });
process.env.EXACT_APP_DIR = resolve(repo, 'examples/t3-code');
const { open } = await import(resolve(repo, 'scripts/agent.mjs'));
const lane = await import(resolve(MINE, 'examples/t3-code/tools/github-lane/lane.mjs'));
const lib = await import(resolve(MINE, 'examples/t3-code/tools/github-lane/lib.mjs'));
const paths = lib.lanePaths();
const origin = 'http://127.0.0.1:16760';
const REPO = 'daehyeonmun2021/playground', SECOND_PR = 158, PRIMARY_PR = 159, SEEDED = 115;
const tag = `${label}-${mode}`;
const rec = (line) => { const text = `${new Date().toISOString()} [${tag}] ${line}`; appendFileSync(resolve(out, 'record.txt'), text + '\n'); console.log(text); };
const gh = (account, args) => { const r = lib.api(paths, account, args); return r; };
const s = await open({ host: 'macos', size: [1280, 840], env: { HOME: resolve(L, `apphome-${label}`) }, storage: `prw-${label}`, epoch: Date.now(),
  onProcess: (p) => { writeFileSync(resolve(L, `${tag}-app.pid`), String(p.pid)); rec(`app pid ${p.pid}`); } });
const nodes = async () => (await s.tree()).nodes;
const ids = async () => (await nodes()).map((n) => n.props?.testId).filter(Boolean);
const has = async (id) => (await ids()).includes(id);
const settle = async (ms = 600) => { await s.clock(`+${ms} real`); await s.clock('settle'); };
const tapIf = async (id, opts) => { if (await has(id)) { await s.tap(id, opts); await settle(); return true; } rec(`(no ${id})`); return false; };
const shot = async (name) => { await s.screenshot(resolve(out, `${tag}-${name}.png`), true); rec(`shot ${name}`); };
const texts = async (target) => {
  try { const t = await s.tree(target); return t.nodes.flatMap((n) => [n.props?.text, n.props?.value, n.props?.['aria-label'], n.props?.placeholder].filter((v) => typeof v === 'string' && v.trim())).map((v) => v.trim()); }
  catch (e) { return [`(no ${target}: ${e.message.slice(0, 80)})`]; }
};
const dump = async (name, target) => { const t = [...new Set(await texts(target))]; writeFileSync(resolve(out, `${tag}-${name}.txt`), t.join('\n')); rec(`${name}: ${t.slice(0, 50).join(' | ').slice(0, 1200)}`); return t; };
const node = async (id) => (await nodes()).find((n) => n.props?.testId === id);
const prop = async (id, key) => (await node(id))?.props?.[key];
const toastsNow = async () => (await ids()).filter((id) => id.startsWith('toast-')).length ? await dump('toasts', 'toast-region').catch(() => []) : [];
const clearToasts = async () => { for (const id of (await ids()).filter((id) => id.startsWith('toast-dismiss-'))) { try { await s.tap(id); await settle(300); } catch {} } };
const openPr = async (n) => {
  await s.type('pull-requests-search', `#${n}`); await settle(3500);
  if (!(await has(`pr-row-${n}`))) await settle(4000);
  await s.tap(`pr-row-${n}`); await settle(5000); await clearToasts();
};
const TRACE = resolve(MINE, 'target/github-lane/servers/primary/t3home/userdata/logs/server.trace.ndjson');
const rpcCounts = () => { const counts = {}; try { for (const line of readFileSync(TRACE, 'utf8').split('\n')) { const m = /"name":"ws\.rpc\.(pullRequests\.[a-zA-Z]+)"/.exec(line); if (m) counts[m[1]] = (counts[m[1]] ?? 0) + 1; } } catch {} return counts; };
const delta = (a, b) => Object.fromEntries(Object.keys(b).filter((k) => (b[k] ?? 0) !== (a[k] ?? 0)).map((k) => [k, (b[k] ?? 0) - (a[k] ?? 0)]));
const logOffset = () => { try { return statSync(paths.ghLog).size; } catch { return 0; } };
const ghSince = (offset) => { try { return readFileSync(paths.ghLog, 'utf8').slice(offset).split('\n').filter(Boolean).map((line) => line.split('\t').slice(-1)[0].trim().replaceAll(REPO, 'O/R').slice(0, 160)).filter((l) => !/graphql.*(pullRequest\(number|viewerPermission|statusCheckRollup|reviewThreads|search\()/.test(l)); } catch { return []; } };
/** Run a step: what it pressed and typed, the server RPCs and gh calls it caused, and its read-back. */
const step = async (name, body) => {
  const before = rpcCounts(), offset = logOffset();
  try { await body(); } catch (e) { rec(`STEP ${name} FAILED: ${e.message.slice(0, 400)}`); try { await shot(`fail-${name}`); } catch {} }
  rec(`  ${name}: server RPCs ${JSON.stringify(delta(before, rpcCounts()))}; gh writes ${JSON.stringify(ghSince(offset).filter((l) => /-X|--method|comment|review|label|reaction|edit|close|reopen|Reaction|update/i.test(l)).slice(0, 12))}`);
};
const lastComment = (n) => { const list = gh('primary', [`repos/${REPO}/issues/${n}/comments?per_page=100`]).body ?? []; return list.at(-1) ?? null; };

try {
  await settle(2000);
  if (await has('welcome-pairing-link')) { await s.type('welcome-pairing-link', `${origin}/pair#token=${lane.pairingToken('primary')}`); await s.tap('welcome-pair'); await settle(5000); }
  for (const id of ['welcome-continue', 'welcome-agents-continue', 'welcome-skip-import']) if (await has(id)) { await s.tap(id); await settle(); }
  await settle(3000); await clearToasts();
  rec(`connected: ${(await has('open-pull-requests')) ? 'sidebar shows Pull Requests' : 'no Pull Requests entry'}`);
  const atStart = rpcCounts();
  await s.tap('open-pull-requests'); await settle(5000); await clearToasts();

  // ── #158 (the second account's): the summary's new controls ──
  await openPr(SECOND_PR);
  await shot('158-summary');
  await dump('158-summary', 'pull-request-panel');
  if (mode === 'before') {
    await s.tap('pull-request-description', { hover: true }); await settle(300); await shot('158-hover-description');
    await openPr(PRIMARY_PR); await s.tap('pull-request-title', { hover: true }).catch(() => {}); await settle(300); await shot('159-summary'); await dump('159-summary', 'pull-request-panel');
    await openPr(SEEDED); await shot('115-summary');
    await s.tap('pull-request-tab-timeline'); await settle(800);
    const group = (await ids()).find((id) => id.startsWith('pull-request-timeline-group-toggle-'));
    if (group) { await s.tap(group); await settle(600); }
    await shot('115-timeline-reactions');
    await s.resize(840, 620); await settle(1000); await s.tap('pull-request-tab-summary'); await settle(600); await shot('115-narrow');
    await openPr(SECOND_PR); await shot('158-narrow');
  } else {
    await s.tap('pull-request-description', { hover: true }); await settle(300); await shot('158-hover-description');

    // ── C1: comment (⌘↵), the exact bytes ──
    const words = 'Does `y` count as a vowel here?\n\nI think it should not — "rhythm" has none.';
    let posted = null;
    await step('comment', async () => {
      await s.tap('pull-request-composer-trigger'); await settle(800); await shot('composer-comment');
      await dump('composer-open', 'pull-request-composer');
      await s.type('pull-request-comment-text', words); await settle(300);
      await s.type('pull-request-comment-text', { key: 'Meta+Enter' }); await settle(300); await shot('composer-posting');
      await settle(6000);
      posted = lastComment(SECOND_PR);
      rec(`read back #${SECOND_PR}: last comment by ${posted?.user?.login}, body equals typed: ${posted?.body === words}, node ${posted?.node_id}`);
      rec(`composer after: open=${await has('pull-request-composer')}`);
      await shot('comment-posted');
    });
    // ── C2: the comment edited in the Write/Preview editor ──
    await step('comment-edit', async () => {
      const card = `pr-page-scroll-s-comment-${posted.node_id}`;
      await s.tap(`pull-request-comment-${posted.node_id}`, { hover: true }); await settle(300); await shot('comment-hover-pencil');
      await s.tap(`${card}-edit`); await settle(600);
      await s.type(`${card}-text`, `${words}\n\nEdited: a **second** thought.`); await settle(300); await shot('comment-editor');
      await s.tap(`${card}-preview`); await settle(2500); await shot('comment-preview');
      await dump('comment-preview', `${card}-preview-body`);
      await s.tap(`${card}-write`); await settle(400);
      await s.type(`${card}-text`, { key: 'Meta+Enter' }); await settle(6000);
      const back = gh('primary', [`repos/${REPO}/issues/comments/${posted.id}`]).body;
      rec(`read back: comment body now ends with the edit: ${back?.body?.endsWith('Edited: a **second** thought.')}`);
      await shot('comment-edited');
      // Escape cancels an open editor; nothing is sent.
      await s.tap(`pull-request-comment-${posted.node_id}`, { hover: true }); await s.tap(`${card}-edit`); await settle(400);
      await s.type(`${card}-text`, { key: 'Escape' }); await settle(400);
      rec(`after Escape the editor is ${await has(`${card}-text`) ? 'still open' : 'closed'}`);
    });
    // ── C6: reactions on the posted comment (picker add, pill remove) ──
    await step('reactions-own', async () => {
      await s.tap(`pull-request-reaction-add-${posted.node_id}`); await settle(500); await shot('reaction-picker');
      await s.tap(`pull-request-reaction-choose-${posted.node_id}-heart`); await settle(200); await shot('reaction-optimistic');
      await settle(5000);
      const r1 = gh('primary', [`repos/${REPO}/issues/comments/${posted.id}/reactions`]).body ?? [];
      rec(`read back: reactions ${JSON.stringify(r1.map((r) => [r.content, r.user?.login]))}; pill pressed ${await prop(`pull-request-reaction-${posted.node_id}-heart`, 'aria-pressed')}`);
      await s.tap(`pull-request-reaction-${posted.node_id}-heart`); await settle(5000);
      const r2 = gh('primary', [`repos/${REPO}/issues/comments/${posted.id}/reactions`]).body ?? [];
      rec(`read back after the pill press: ${JSON.stringify(r2.map((r) => [r.content, r.user?.login]))}`);
    });
    // ── C3: labels (a name with a space and a slash), applied and removed ──
    await step('labels', async () => {
      await s.tap('pull-request-labels-picker'); await settle(300); await shot('labels-loading');
      await settle(4000); await shot('labels-menu'); await dump('labels-menu', 'pull-request-labels-menu');
      await s.type('pull-request-labels-search', 'docs'); await settle(400); await shot('labels-search');
      await s.tap('pull-request-labels-option-area/docs and help'); await settle(5000);
      rec(`read back labels: ${JSON.stringify((gh('primary', [`repos/${REPO}/issues/${SECOND_PR}/labels`]).body ?? []).map((l) => l.name))}; menu still open: ${await has('pull-request-labels-menu')}`);
      await shot('labels-applied');
      await s.tap('pull-request-labels-option-area/docs and help'); await settle(5000);
      rec(`read back after removing: ${JSON.stringify((gh('primary', [`repos/${REPO}/issues/${SECOND_PR}/labels`]).body ?? []).map((l) => l.name))}`);
      await s.type('pull-request-labels-search', 'zzz'); await settle(300); await dump('labels-nomatch', 'pull-request-labels-menu');
      await s.type('pull-request-labels-search', { key: 'Escape' }); await settle(400);
      rec(`after Escape the label menu is ${await has('pull-request-labels-menu') ? 'open' : 'closed'}`);
    });
    // ── C1: review verdicts with a summary (reviewer on #158) ──
    for (const [verdict, words2, state] of [['comment', 'A first pass: reads well.', 'COMMENTED'], ['approve', 'Approved: the counter is right.', 'APPROVED'], ['request-changes', 'Please handle uppercase vowels too.', 'CHANGES_REQUESTED']]) {
      await step(`review-${verdict}`, async () => {
        await s.tap('pull-request-composer-trigger'); await settle(600);
        if (await has('pull-request-composer-mode-review')) await s.tap('pull-request-composer-mode-review'); await settle(300);
        await s.type('pull-request-review-summary', words2); await settle(200);
        await s.type('pull-request-review-verdict', verdict); await settle(300); await shot(`review-${verdict}`);
        await dump(`review-${verdict}-form`, 'pull-request-review-form');
        await s.tap('pull-request-review-submit'); await settle(6000);
        const reviews = gh('primary', [`repos/${REPO}/pulls/${SECOND_PR}/reviews?per_page=100`]).body ?? [];
        const last = reviews.at(-1);
        rec(`read back: last review ${last?.state} by ${last?.user?.login}, body equals: ${last?.body === words2} (want ${state})`);
        await dump(`review-${verdict}-toast`, 'toast-region').catch(() => {});
        await clearToasts();
      });
    }
    // Close with comment, then reopen with comment.
    await step('close-with-comment', async () => {
      await s.tap('pull-request-composer-trigger'); await settle(600);
      if (await has('pull-request-composer-mode-comment')) await s.tap('pull-request-composer-mode-comment');
      await s.type('pull-request-comment-text', 'Closing for now; will reopen.'); await settle(200); await shot('close-with-comment');
      await s.tap('pull-request-comment-close'); await settle(8000);
      const pr = gh('primary', [`repos/${REPO}/pulls/${SECOND_PR}`]).body;
      rec(`read back: state ${pr?.state}; last comment ${JSON.stringify(lastComment(SECOND_PR)?.body)}`);
      await shot('closed'); await clearToasts();
      await s.tap('pull-request-composer-trigger'); await settle(600);
      await s.type('pull-request-comment-text', 'Reopening.'); await settle(200); await shot('reopen-with-comment');
      await s.tap('pull-request-comment-reopen'); await settle(8000);
      rec(`read back: state ${gh('primary', [`repos/${REPO}/pulls/${SECOND_PR}`]).body?.state}; last comment ${JSON.stringify(lastComment(SECOND_PR)?.body)}`);
      await clearToasts();
    });
    // ── C2: a 40-line comment collapses at 240pt ──
    let long = null;
    await step('collapse', async () => {
      const body = Array.from({ length: 40 }, (_, i) => `Line ${i + 1} of a long note about vowels.`).join('\n\n');
      long = gh('second', ['-X', 'POST', `repos/${REPO}/issues/${SECOND_PR}/comments`, '-f', `body=${body}`]).body;
      rec(`posted a 40-line comment as the second account: ${long?.node_id}`);
      await s.tap('pull-request-more'); await settle(300); await s.tap('pr-more-refresh'); await settle(7000);
      const frame = `pr-page-scroll-s-comment-body-${long.node_id}`;
      await s.tap(frame, { hover: true }).catch(() => {}); await settle(300); await shot('collapse-collapsed');
      rec(`toggle: ${await prop(`${frame}-toggle`, 'aria-expanded')} ${JSON.stringify(await texts(`${frame}-toggle`))}`);
      await s.tap(`${frame}-toggle`); await settle(600); await shot('collapse-expanded');
      rec(`toggle after: ${await prop(`${frame}-toggle`, 'aria-expanded')} ${JSON.stringify(await texts(`${frame}-toggle`))}`);
      await s.tap(`${frame}-toggle`); await settle(600);
    });
    // ── #115: the seeded comment with two reactions (join the heart, leave it) ──
    await step('reactions-seeded', async () => {
      await openPr(SEEDED);
      const list = gh('primary', [`repos/${REPO}/issues/${SEEDED}/comments?per_page=100`]).body ?? [];
      const thanks = list.find((c) => c.body?.includes('<!-- ref:comment-1 -->'));
      const pill = `pull-request-reaction-${thanks.node_id}-heart`;
      rec(`seeded comment ${thanks.node_id}: pills ${JSON.stringify(await texts(`pull-request-reactions-${thanks.node_id}`))}`);
      await s.tap(pill, { hover: true }); await settle(1200); await shot('115-reaction-tooltip');
      await s.tap(pill); await settle(200);
      rec(`optimistic: ${await prop(pill, 'aria-label')} pressed ${await prop(pill, 'aria-pressed')}`);
      await settle(5000);
      const r1 = gh('primary', [`repos/${REPO}/issues/comments/${thanks.id}/reactions`]).body ?? [];
      rec(`read back: ${JSON.stringify(r1.map((r) => [r.content, r.user?.login]))}`);
      await s.tap(pill, { hover: true }); await settle(1200); await shot('115-reaction-joined');
      await dump('115-reaction-tooltip', `pull-request-reactions-${thanks.node_id}`);
      await s.tap(pill); await settle(5000);
      const r2 = gh('primary', [`repos/${REPO}/issues/comments/${thanks.id}/reactions`]).body ?? [];
      rec(`read back after leaving: ${JSON.stringify(r2.map((r) => [r.content, r.user?.login]))}`);
      await s.tap('pull-request-tab-timeline'); await settle(800);
      const group = (await ids()).find((id) => id.startsWith('pull-request-timeline-group-toggle-'));
      if (group) { await s.tap(group); await settle(600); }
      await shot('115-timeline-reactions');
      await s.tap('pull-request-tab-summary'); await settle(600);
    });
    // ── #159 (the primary's own): the author's composer, title and description, reviewers ──
    await openPr(PRIMARY_PR);
    await step('author-composer', async () => {
      await s.tap('pull-request-composer-trigger'); await settle(600);
      if (await has('pull-request-composer-mode-review')) await s.tap('pull-request-composer-mode-review'); await settle(300);
      await shot('author-review'); await dump('author-review', 'pull-request-review-form');
      rec(`submit enabled with an empty summary: ${!(await prop('pull-request-review-submit', 'disabled'))}`);
      await s.type('pull-request-review-summary', { key: 'Escape' }); await settle(400);
      rec(`after Escape the composer is ${await has('pull-request-composer') ? 'open' : 'closed'}`);
    });
    await step('title', async () => {
      const original = gh('primary', [`repos/${REPO}/pulls/${PRIMARY_PR}`]).body.title;
      await s.tap('pull-request-title', { hover: true }); await settle(300); await shot('title-hover');
      await s.tap('pull-request-edit-title'); await settle(400); await shot('title-editor');
      const before = rpcCounts();
      await s.type('pull-request-title-input', { key: 'Enter' }); await settle(1500);
      rec(`unchanged title: pullRequests.update calls ${(rpcCounts()['pullRequests.update'] ?? 0) - (before['pullRequests.update'] ?? 0)}`);
      await s.tap('pull-request-edit-title'); await settle(400);
      await s.type('pull-request-title-input', '  Describe the vowel counter (edited)  '); await settle(200);
      await s.type('pull-request-title-input', { key: 'Enter' }); await settle(5000);
      rec(`read back title: ${JSON.stringify(gh('primary', [`repos/${REPO}/pulls/${PRIMARY_PR}`]).body.title)}`);
      await shot('title-saved');
      await s.tap('pull-request-edit-title'); await settle(400);
      await s.type('pull-request-title-input', original); await s.type('pull-request-title-input', { key: 'Enter' }); await settle(5000);
      rec(`restored title: ${JSON.stringify(gh('primary', [`repos/${REPO}/pulls/${PRIMARY_PR}`]).body.title)}`);
    });
    await step('description', async () => {
      const original = gh('primary', [`repos/${REPO}/pulls/${PRIMARY_PR}`]).body.body;
      await s.tap('pull-request-description', { hover: true }); await settle(300);
      await s.tap('pr-page-scroll-description-edit'); await settle(500);
      await s.type('pr-page-scroll-description-text', '## Counting vowels\n\n- `a e i o u`\n- not `y`'); await settle(200); await shot('description-editor');
      await s.tap('pr-page-scroll-description-preview'); await settle(2500); await shot('description-preview');
      await dump('description-preview', 'pr-page-scroll-description-preview-body');
      await s.tap('pr-page-scroll-description-write'); await settle(300);
      await s.type('pr-page-scroll-description-text', { key: 'Meta+Enter' }); await settle(6000);
      rec(`read back body: ${JSON.stringify(gh('primary', [`repos/${REPO}/pulls/${PRIMARY_PR}`]).body.body)}`);
      await shot('description-saved');
      await s.tap('pull-request-description', { hover: true }); await s.tap('pr-page-scroll-description-edit'); await settle(500);
      await s.type('pr-page-scroll-description-text', original ?? ''); await s.tap('pr-page-scroll-description-save'); await settle(6000);
      rec(`restored body equals original: ${gh('primary', [`repos/${REPO}/pulls/${PRIMARY_PR}`]).body.body === original}`);
    });
    await step('reviewers', async () => {
      await s.tap('pull-request-reviewers-picker'); await settle(300); await shot('reviewers-loading');
      await settle(5000); await shot('reviewers-menu'); await dump('reviewers-menu', 'pull-request-reviewers-menu');
      const option = (await ids()).find((id) => id.startsWith('pull-request-reviewers-option-'));
      await s.tap(option); await settle(5000);
      rec(`read back requested: ${JSON.stringify((gh('primary', [`repos/${REPO}/pulls/${PRIMARY_PR}/requested_reviewers`]).body?.users ?? []).map((u) => u.login))}; faces ${JSON.stringify(await texts('pull-request-reviewers'))}`);
      await shot('reviewers-asked'); await dump('reviewers-toast', 'toast-region').catch(() => {});
      await s.tap(option); await settle(5000);
      rec(`read back after taking it back: ${JSON.stringify((gh('primary', [`repos/${REPO}/pulls/${PRIMARY_PR}/requested_reviewers`]).body?.users ?? []).map((u) => u.login))}`);
      await s.type('pull-request-reviewers-search', { key: 'Escape' }); await settle(300);
      await clearToasts();
    });
    // The narrow window: the composer and a picker at 840×620.
    await step('narrow', async () => {
      await s.resize(840, 620); await settle(1000);
      await s.tap('pull-request-composer-trigger'); await settle(600); await shot('narrow-composer');
      await s.type('pull-request-comment-text', { key: 'Escape' }); await settle(300);
      await s.tap('pull-request-labels-picker'); await settle(4000); await shot('narrow-labels');
      await s.type('pull-request-labels-search', { key: 'Escape' }); await settle(300);
    });
    // Cleanup: the test comments the drive made on #158 (the reviews stay: GitHub keeps submitted reviews).
    for (const c of [posted, long].filter(Boolean)) {
      const account = c === long ? 'second' : 'primary';
      rec(`deleted comment ${c.id}: HTTP ${gh(account, ['-X', 'DELETE', `repos/${REPO}/issues/comments/${c.id}`]).status}`);
    }
    for (const c of (gh('primary', [`repos/${REPO}/issues/${SECOND_PR}/comments?per_page=100`]).body ?? []).filter((c) => ['Closing for now; will reopen.', 'Reopening.'].includes(c.body))) {
      rec(`deleted comment ${c.id}: HTTP ${gh('primary', ['-X', 'DELETE', `repos/${REPO}/issues/comments/${c.id}`]).status}`);
    }
  }
  rec(`server RPCs this session: ${JSON.stringify(delta(atStart, rpcCounts()))}`);
  const logs = await s.logs();
  writeFileSync(resolve(out, `${tag}-logs.json`), JSON.stringify(logs, null, 1));
  rec(`log lines ${logs.lines.length}; errors: ${logs.lines.filter((l) => /error|refused|failed/i.test(JSON.stringify(l))).length}`);
} catch (e) { rec(`ERROR ${e.stack?.slice(0, 1200)}`); try { await shot('error'); writeFileSync(resolve(out, `${tag}-error-tree.json`), JSON.stringify(await s.tree(), null, 1)); } catch {} }
finally { await s.close?.(); }
process.exit(0);
