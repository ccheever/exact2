BAR, GAP, HEAD = 46, 12, 52
def font(size):
    for path in ('/System/Library/Fonts/SFNS.ttf', '/System/Library/Fonts/Helvetica.ttc'):
        try: return ImageFont.truetype(path, size)
        except Exception: pass
    return ImageFont.load_default()
def cell(image, label, w, h):
    out = Image.new('RGB', (w, h + BAR), '#f4f4f5')
    d = ImageDraw.Draw(out); d.rectangle([0, 0, w - 1, BAR - 1], fill='#18181b'); d.text((12, 11), label, fill='white', font=font(20))
    image = image.convert('RGB'); image.thumbnail((w, h), Image.LANCZOS)
    out.paste(image, (0, BAR)); return out
def grid(name, title, cells, w, h):
    out = Image.new('RGB', (len(cells) * w + (len(cells) - 1) * GAP, HEAD + h + BAR), 'white')
    ImageDraw.Draw(out).text((12, 12), title, fill='#18181b', font=font(26))
    for i, (image, label) in enumerate(cells): out.paste(cell(image, label, w, h), (i * (w + GAP), HEAD))
    out.save(f'{OUT}/{name}', optimize=True); print(name, out.size)

def textcard(lines, w, h, size=26):
    im = Image.new('RGB', (w, h), 'white'); d = ImageDraw.Draw(im); y = 24
    for line in lines:
        d.text((24, y), line, fill='#18181b', font=font(size)); y += int(size * 1.6)
    return im
