// Feasapp: rem probe for the T3 interface font size task.
shape Greeting
  text: string

component Feasapp
  resource greeting = greeting("Feasapp") as shape Greeting
  state big = false
  action pick(v: number)
    setRootFontSize(v)
  view
    column testId="page" gap="0.5rem" padding="1rem" font-size="0.875rem" line-height="1.25rem" position="relative" width="100%" height="100%"
      row gap=8
        button press=pick(12) testId="s12"
          text "12"
        button press=pick(16) testId="s16"
          text "16"
        button press=pick(20) testId="s20"
          text "20"
      box testId="box" width="10rem" height="2rem" min-width="4rem" padding-left="0.5rem" margin-top="0.25rem" border-radius="0.625rem" border-width=1 border-style="solid"
      box testId="pos" position="absolute" top="1rem" left="2rem" width="2rem" height="1rem"
      svg width="1rem" height="1rem" viewBox="0 0 24 24" testId="svg"
        path d="m4 17 6-5-6-5"
      Probe(size=14)
      box testId="calc" width=`calc(${big ? 10 : 90}px + 2.5rem)` height="calc(1rem + 4px)" padding-left="calc(90px + 2.5rem)"
      text greeting.text testId="t" font-size="0.75rem" line-height="1rem" letter-spacing="0.025em"
      row testId="gaprow" column-gap="0.375rem"
        box width=10 height=10 testId="g1"
        box width=10 height=10 testId="g2"

component Probe
  props
    size: number
  view
    svg width=`${size / 16}rem` height=`${size / 16}rem` viewBox="0 0 24 24" testId="icon"
      path d="m4 17 6-5-6-5"
