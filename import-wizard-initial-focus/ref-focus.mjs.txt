// import-wizard-initial-focus: where the reference's browser import wizard puts the focus when it opens and after each
// step change, and the profile removal confirm. Lane fixture homes only (Brave held by a foreign-host SingletonLock,
// Firefox "broken" profile with a garbage cookies.sqlite). Firefox imports only (no Keychain read).
// Usage: node ref-focus.mjs <cdpPort> <out.json> <shotDir>. Never closes the browser.
import { createRequire } from 'node:module';
import fs from 'node:fs';
const require = createRequire('/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-ref/src-1e2ecbd975/apps/desktop/package.json');
const { chromium } = require('playwright-core');
const [port, outFile, shots] = process.argv.slice(2);
const browser = await chromium.connectOverCDP(`http://127.0.0.1:${port}`);
const page = browser.contexts().flatMap((c) => c.pages()).find((p) => !p.url().startsWith('devtools://') && !p.url().startsWith('data:'));
await page.setViewportSize({ width: 1280, height: 840 });
const log = [];
const state = async (label) => {
  const s = await page.evaluate(() => {
    const a = document.activeElement;
    const dialog = [...document.querySelectorAll('[role=dialog],[role=alertdialog]')].find((d) => d.getAttribute('data-slot') === 'dialog-popup') ?? null;
    const describe = (e) => {
      if (!e) return null;
      if (e === document.body) return 'BODY';
      if (e === dialog) return `${e.nodeName}[role=${e.getAttribute('role')}] (the popup itself)`;
      const label = e.getAttribute('aria-label');
      const text = (e.innerText ?? '').replace(/\s+/g, ' ').trim().slice(0, 48);
      return `${e.nodeName}${label ? `[aria-label=${label}]` : ''}${e.getAttribute('aria-pressed') ? `[aria-pressed=${e.getAttribute('aria-pressed')}]` : ''} "${text}"`;
    };
    const tabbables = dialog ? [...dialog.querySelectorAll('button:not([disabled]),[tabindex]:not([tabindex="-1"]),input,a[href]')].map((e) => describe(e)) : [];
    const r = a && a !== document.body ? a.getBoundingClientRect() : null;
    return { rect: r ? { x: r.x, y: r.y, w: r.width, h: r.height } : null, active: describe(a), focusVisible: a?.matches?.(':focus-visible') ?? false, insideDialog: dialog ? dialog.contains(a) : null,
      dialogTitle: dialog?.querySelector('h2')?.innerText ?? null, dialogTabbables: tabbables };
  });
  log.push({ label, ...s });
  console.log(JSON.stringify({ label, ...s }));
};
const click = async (locator) => { const b = await locator.boundingBox(); await page.mouse.click(b.x + b.width / 2, b.y + b.height / 2); };
const dialog = () => page.locator('[role=dialog]');
const openSource = async (name) => {
  await click(page.getByRole('button', { name: 'Add profile' }));
  await page.waitForTimeout(700);
  await click(page.getByRole('menuitem', { name }));
  await page.waitForTimeout(1500);
};
const shot = async (name) => { if (shots) await page.screenshot({ path: `${shots}/${name}.png` }); };

await page.goto(new URL('#/settings/integrations', page.url()).href);
await page.waitForTimeout(2500);

// 1. Chrome opens on Configure (pointer: Add profile, then the menu row).
await openSource('Chrome');
await state('Chrome: wizard open on Configure (pointer)');
await shot('ref-chrome-configure-open');
await page.keyboard.press('Tab');
await state('Chrome Configure: Tab');
await page.keyboard.press('Shift+Tab');
await page.keyboard.press('Shift+Tab');
await state('Chrome Configure: Shift+Tab twice');
await click(dialog().getByRole('button', { name: 'Cancel' }));
await page.waitForTimeout(800);
await state('Chrome: Cancel by pointer');

// 2. Chrome opened from the keyboard: focus Add profile, Enter, ArrowDown to Chrome, Enter.
await page.getByRole('button', { name: 'Add profile' }).focus();
await page.keyboard.press('Enter');
await page.waitForTimeout(700);
await state('Add profile menu opened by Enter');
await page.getByRole('menuitem', { name: 'Chrome' }).focus();
await page.keyboard.press('Enter');
await page.waitForTimeout(1500);
await state('Chrome: wizard open on Configure (keyboard)');
await page.keyboard.press('Escape');
await page.waitForTimeout(800);
await state('Chrome: Escape');

// 3. Brave is running (SingletonLock): opens on the quit step.
await openSource('Brave');
await state('Brave: wizard open on Quit');
await shot('ref-brave-quit-open');
await click(dialog().getByRole('button', { name: 'I’ve quit it' }));
await state('Brave: right after I’ve quit it (pointer)');
await page.waitForTimeout(1500);
await state('Brave: after the recheck (still running: Quit again)');
await shot('ref-brave-quit-again');
await page.keyboard.press('Tab');
await state('Brave Quit again: Tab');
await page.keyboard.press('Escape');
await page.waitForTimeout(800);

// 4. Firefox: the broken profile's import fails (blocked), Try again, then Close.
await openSource('Firefox');
await state('Firefox: wizard open on Configure');
await click(dialog().getByRole('button', { name: /broken/ }));
await page.waitForTimeout(300);
await state('Firefox: broken tile chosen (pointer)');
await click(dialog().getByRole('button', { name: 'Import', exact: true }));
await state('Firefox: right after Import (pointer)');
await page.waitForTimeout(2500);
await state('Firefox: after the import (blocked)');
await shot('ref-firefox-blocked');
const retry = dialog().getByRole('button', { name: 'Try again' });
if (await retry.count()) {
  await click(retry);
  await state('Firefox: right after Try again (pointer)');
  await page.waitForTimeout(2500);
  await state('Firefox: after Try again');
}
await page.keyboard.press('Tab');
await state('Firefox blocked: Tab');
await click(dialog().getByRole('button', { name: 'Close', exact: true }).first());
await page.waitForTimeout(800);
await state('Firefox: Close by pointer');

// 5. Firefox by keyboard: the default profile into New profile; Tab to Import, Enter; done; Done by Enter.
await openSource('Firefox');
await state('Firefox 2: wizard open on Configure');
for (let i = 0; i < 6; i++) {
  const s = await page.evaluate(() => (document.activeElement?.innerText ?? '').trim());
  if (s === 'Import') break;
  await page.keyboard.press('Tab');
}
await state('Firefox 2: Tab to Import');
await page.keyboard.press('Enter');
await state('Firefox 2: right after Enter on Import');
await page.waitForTimeout(3000);
await state('Firefox 2: done');
await shot('ref-firefox-done');
await page.keyboard.press('Tab');
await state('Firefox 2 done: Tab');
await page.keyboard.press('Escape');
await page.waitForTimeout(800);
await state('Firefox 2: Escape on done');

fs.writeFileSync(outFile, JSON.stringify(log, null, 2));
process.exit(0);
