import { Database } from 'bun:sqlite';
import * as crypto from 'node:crypto';
import * as fs from 'node:fs';
const BH = process.argv[2];

const mk = p => fs.mkdirSync(p, { recursive: true });
const AS = `${BH}/Library/Application Support`;
const key = secret => crypto.pbkdf2Sync(secret, 'saltysalt', 1003, 16, 'sha1');
const v10 = (plain, k) => { const c = crypto.createCipheriv('aes-128-cbc', k, Buffer.alloc(16, 0x20)); return Buffer.concat([Buffer.from('v10'), c.update(plain), c.final()]); };
const bound = (host, value) => Buffer.concat([crypto.createHash('sha256').update(host).digest(), Buffer.from(value)]);
const far = (Date.now() / 1000 + 30 * 86400 + 11644473600) * 1e6; // Chromium µs since 1601, 30 days out
function chromium(file, version, secret, rows) {
  mk(file.slice(0, file.lastIndexOf('/')));
  const db = new Database(file), k = key(secret);
  db.run('create table meta (key text primary key, value text not null)');
  db.run("insert into meta values ('version', ?)", [String(version)]);
  db.run(`create table cookies (host_key text not null, name text not null, value text not null, encrypted_value blob not null, path text not null, expires_utc integer not null,
    is_secure integer not null, is_httponly integer not null, samesite integer not null, top_frame_site_key text not null default '')`);
  for (const r of rows) {
    const plain = version >= 24 ? bound(r.bindHost ?? r.host, r.value) : Buffer.from(r.value);
    db.run('insert into cookies values (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)', [r.host, r.name, '', v10(plain, k), r.path ?? '/', Math.round(far), r.secure ?? 0, r.httpOnly ?? 0, r.sameSite ?? 1, r.top ?? '']);
  }
  db.close();
}
