---- restore-probe.ts
// Restore defaults' labels for frame rate 60, zoom 125% and iPhone 12 Pro (the review's scenario), on a given checkout.
const dir = process.argv[2];
const core = await import(`${dir}/settings-core.ts`);
const defaults = await import(`${dir}/browser-defaults.ts`);
const local = { deviceSettings: { composerCollapseOnScroll: true, planModeEnabled: false, timestampFormat: 'locale', appearanceMode: 'system', sendShortcut: 'enter' }, clientSettings: core.decodeClientPrefs({}), groupingMode: 'repository' };
const owner = { local };
defaults.applyBrowserDefault(owner, 'frame-rate', '60');
defaults.applyBrowserDefault(owner, 'zoom', '1.25');
defaults.applyBrowserDefault(owner, 'viewport', 'iphone-12-pro');
const labels = core.restoreLabels(local, {}, false);
console.log(`restoreLabels: ${JSON.stringify(labels)} (restoreCount ${labels.length}: Restore ${labels.length ? 'enabled' : 'disabled'})`);
core.restoreDeviceDefaults(local);
const view = defaults.browserDefaultsView(owner);
console.log(`after restoreDeviceDefaults: viewport ${view.viewportLabel}, zoom ${view.zoomLabel}, appearance ${view.appearanceLabel}, frame rate ${view.frameRateLabel}`);
const menu = view.viewportOptions.slice(0, 4).map(o => JSON.stringify(o)).join('\n  ');
console.log(`viewport menu rows (first four):\n  ${menu}`);
