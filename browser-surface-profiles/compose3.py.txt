def font(size, bold=False):
    for f in (["/System/Library/Fonts/SFNS.ttf"] if not bold else []) + ["/System/Library/Fonts/Helvetica.ttc"]:
        try:
            return ImageFont.truetype(f, size)
        except Exception:
            pass
    return ImageFont.load_default()


def wrap(draw, text, fnt, width):
    lines = []
    for para in text.split("\n"):
        line = ""
        for word in para.split(" "):
            trial = (line + " " + word).strip()
            if draw.textlength(trial, font=fnt) <= width:
                line = trial
            else:
                lines.append(line)
                line = word
        lines.append(line)
    return lines


def cell(spec):
    canvas = Image.new("RGB", (W, H), (252, 252, 253))
    d = ImageDraw.Draw(canvas)
    if spec is None:
        d.text((24, H // 2 - 12), "(no such step here)", fill=(150, 150, 158), font=font(20))
        return canvas
    if isinstance(spec, tuple):
        fnt = font(19)
        y = 22
        for line in wrap(d, spec[1], fnt, W - 48):
            d.text((24, y), line, fill=(40, 40, 46), font=fnt)
            y += 27
        return canvas
    im = Image.open(spec).convert("RGB")
    return im.resize((W, int(im.height * W / im.width)))


def compose(name, caption, rows, heads=(BEFORE, AFTER, REF)):
    pad, head, cap, rowcap = 20, 44, 52, 30
    width = 3 * W + 4 * pad
    height = cap + head + sum(rowcap + H + pad for _ in rows) + pad
    canvas = Image.new("RGB", (width, height), (244, 244, 246))
    d = ImageDraw.Draw(canvas)
    d.text((pad, 14), caption, fill=(24, 24, 28), font=font(24))
    for c, label in enumerate(heads):
        d.text((pad + c * (W + pad), cap + 8), label, fill=(60, 60, 72), font=font(21))
    y = cap + head
    for label, cells in rows:
        d.text((pad, y + 4), label, fill=(90, 90, 100), font=font(18))
        y += rowcap
        for c, spec in enumerate(cells):
            x = pad + c * (W + pad)
            canvas.paste(cell(spec), (x, y))
            d.rectangle([x - 1, y - 1, x + W, y + H], outline=(205, 205, 210))
        y += H + pad
    os.makedirs(OUT, exist_ok=True)
    canvas.save(f"{OUT}/{name}", optimize=True)
    print(name, canvas.size)
