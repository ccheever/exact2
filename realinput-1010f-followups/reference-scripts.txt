// ===== ref-lib.mjs
// Shared helpers for the reference's Code tab gutter (realinput-1010f-followups).
import { createRequire } from 'node:module';
const require = createRequire('/Users/daehyeonmun/orca/workspaces/exact2/t3-code/target/t3-ref/src-1e2ecbd975/apps/desktop/package.json');
export const { chromium } = require('playwright-core');
export async function open(port) {
  const browser = await chromium.connectOverCDP(`http://127.0.0.1:${port}`);
  const page = browser.contexts().flatMap((c) => c.pages()).find((p) => !p.url().startsWith('devtools://') && !p.url().startsWith('data:'));
  await page.setViewportSize({ width: 1280, height: 840 });
  return page;
}
export const log = (o) => console.log(JSON.stringify(o));
/** Every visible gutter number in the diffs' shadow roots: {n, type, index, x, y, w, h, selected}. */
export const gutter = (page) => page.evaluate(() => {
  const out = [];
  for (const c of document.querySelectorAll('diffs-container')) {
    const root = c.shadowRoot; if (!root) continue;
    for (const el of root.querySelectorAll('[data-column-number]')) { const r = el.getBoundingClientRect(); if (r.width > 0 && r.y > 0 && r.y < 840) out.push({ n: el.getAttribute('data-column-number'), type: el.getAttribute('data-line-type'), index: el.getAttribute('data-line-index'), x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height), selected: el.getAttribute('data-selected-line') }); }
  }
  return out;
});
/** The open draft's label, the pending cards and other comment text in light and shadow DOM. */
export const texts = (page, re) => page.evaluate((source) => {
  const re = new RegExp(source); const hits = [];
  const walk = (root) => { for (const el of root.querySelectorAll('*')) { if (el.children.length === 0 && re.test(el.textContent || '')) { const r = el.getBoundingClientRect(); if (r.width > 0) hits.push({ text: (el.textContent || '').trim().slice(0, 80), x: Math.round(r.x), y: Math.round(r.y) }); } if (el.shadowRoot) walk(el.shadowRoot); } };
  walk(document); return hits;
}, re);
export async function scrollToUsage(page) {
  const find = () => page.evaluate(() => { const hits = []; const walk = (root) => { for (const el of root.querySelectorAll('*')) { if (el.children.length === 0 && (el.textContent || '').trim() === 'docs/usage.md') { const r = el.getBoundingClientRect(); if (r.width > 0) hits.push({ x: r.x, y: r.y }); } if (el.shadowRoot) walk(el.shadowRoot); } }; walk(document); return hits[0]; });
  const scroller = await page.evaluateHandle(() => { for (const start of document.querySelectorAll('diffs-container')) { let el = start; while (el && !(el.scrollHeight > el.clientHeight + 10 && /(auto|scroll)/.test(getComputedStyle(el).overflowY))) el = el.parentElement; if (el) return el; } return document.scrollingElement; });
  let hit = await find();
  for (let i = 0; i < 80 && !hit; i++) { await scroller.evaluate(el => { el.scrollTop += 400; }); await page.waitForTimeout(250); hit = await find(); }
  if (hit) { await scroller.evaluate((el, y) => { el.scrollTop += y - 240; }, hit.y); await page.waitForTimeout(800); }
  return find();
}

const page = await open(port);

const start = { x: pill.x + 1.5, y: pill.y + pill.height / 2 };
