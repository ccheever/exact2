// realinput-1010g-followups RG-4: another process's accessibility tree through AXUIElement (the API VoiceOver uses),
// with a guard against elements that loop back to an ancestor. Usage: axwalk <pid> [maxDepth]
import ApplicationServices
import Foundation

let args = CommandLine.arguments
guard args.count >= 2, let pid = Int32(args[1]) else { print("usage: axwalk <pid> [maxDepth]"); exit(2) }
let maxDepth = args.count > 2 ? Int(args[2]) ?? 60 : 60
print("AXIsProcessTrusted", AXIsProcessTrusted())

func copy(_ e: AXUIElement, _ name: String) -> (AXError, CFTypeRef?) {
    var v: CFTypeRef?
    let err = AXUIElementCopyAttributeValue(e, name as CFString, &v)
    return (err, v)
}
func show(_ v: CFTypeRef?) -> String {
    guard let v else { return "nil" }
    if CFGetTypeID(v) == CFStringGetTypeID() { let s = v as! String; return "\"" + (s.count > 80 ? String(s.prefix(80)) + "…" : s) + "\"" }
    if CFGetTypeID(v) == CFURLGetTypeID() { return (v as! NSURL).absoluteString ?? "?" }
    return "\(v)"
}
func line(_ e: AXUIElement) -> String {
    let (re, role) = copy(e, "AXRole")
    guard re == .success, let role = role as? String else { return "<element> AXRole err \(re.rawValue)" }
    var s = role
    if let sub = copy(e, "AXSubrole").1 as? String { s += "/" + sub }
    for a in ["AXTitle", "AXValue", "AXDescription"] {
        let (err, v) = copy(e, a)
        if err == .success, let v, !(CFGetTypeID(v) == CFStringGetTypeID() && (v as! String).isEmpty) { s += " \(a)=\(show(v))" }
    }
    let (ue, url) = copy(e, "AXURL")
    if ue == .success { s += " AXURL=\(show(url))" } else if role == "AXLink" { s += " AXURL=(err \(ue.rawValue))" }
    var actions: CFArray?
    if AXUIElementCopyActionNames(e, &actions) == .success, let names = actions as? [String], !names.isEmpty { s += " actions=\(names.joined(separator: ","))" }
    return s
}
func children(_ e: AXUIElement) -> [AXUIElement] { (copy(e, "AXChildren").1 as? [AXUIElement]) ?? [] }
func dump(_ e: AXUIElement, _ depth: Int, _ path: [AXUIElement]) {
    print(String(repeating: "  ", count: depth) + line(e))
    if depth >= maxDepth { return }
    for c in children(e) {
        if path.contains(where: { CFEqual($0, c) }) || CFEqual(c, e) { print(String(repeating: "  ", count: depth + 1) + "(loops to an ancestor)"); continue }
        dump(c, depth + 1, path + [e])
    }
}
let app = AXUIElementCreateApplication(pid)
var candidates: [AXUIElement] = []
for source in ["AXWindows", "AXChildren"] {
    let found = (copy(app, source).1 as? [AXUIElement]) ?? []
    print("\(source): \(found.map(line))")
    candidates += found
}
for single in ["AXMainWindow", "AXFocusedWindow"] {
    let (err, v) = copy(app, single)
    print("\(single): err \(err.rawValue) \(v.map { line($0 as! AXUIElement) } ?? "nil")")
    if let v { candidates.append(v as! AXUIElement) }
}
var seen: [AXUIElement] = []
for w in candidates where (copy(w, "AXRole").1 as? String) == "AXWindow" && !seen.contains(where: { CFEqual($0, w) }) {
    seen.append(w)
    dump(w, 0, [app])
}
